# 4.24 파일 기반 성적관리

```python
def save_student(name, kor, eng, math):

    with open(
        "students.txt",
        "a",
        encoding="utf-8"
    ) as file:

        file.write(
            f"{name},{kor},{eng},{math}\n"
        )


def show_students():

    with open(
        "students.txt",
        "r",
        encoding="utf-8"
    ) as file:

        for line in file:

            name, kor, eng, math = line.strip().split(",")

            kor = int(kor)
            eng = int(eng)
            math = int(math)

            total = kor + eng + math
            avg = total / 3

            print(
                name,
                total,
                f"{avg:.2f}"
```

```python
def repeat(count):
```

# 8.14 직접 이터레이터 만들기

```python
class Counter:

    def __init__(
        self,
        max_value
    ):

        self.current = 0
        self.max_value = max_value

    def __iter__(self):

        return self

    def __next__(self):

        if self.current >= self.max_value:

            raise StopIteration

        self.current += 1

        return self.current
```

# 8.25 최종 통합 성적관리 프로그램

이제 지금까지 배운 내용을 하나의 프로그램으로 연결해본다.

```python
from dataclasses import dataclass
from functools import wraps
import json


def log(func):

    @wraps(func)
    def wrapper(
        *args,
        **kwargs
    ):

        print(
            f"[실행] {func.__name__}"
        )

        return func(
            *args,
            **kwargs
        )

    return wrapper


@dataclass
class Student:

    name: str
    kor: int
    eng: int
    math: int

    @property
    def total(self) -> int:

        return (
            self.kor
            + self.eng
            + self.math
        )

    @property
    def average(self) -> float:

        return self.total / 3

    @property
    def grade(self) -> str:

        avg = self.average

        if avg >= 90:
            return "A"

        if avg >= 80:
            return "B"

        if avg >= 70:
            return "C"

        if avg >= 60:
            return "D"

        return "F"

    def to_dict(self) -> dict:

        return {

            "name": self.name,

            "kor": self.kor,

            "eng": self.eng,

            "math": self.math
        }


class StudentManager:

    def __init__(self):

        self.students: list[
            Student
        ] = []

    @log
    def add_student(
        self,
        student: Student
    ) -> None:

        self.students.append(
            student
        )

    def find(
        self,
        name: str
    ) -> Student | None:

        for student in self.students:

            if student.name == name:

                return student

        return None

    def students_generator(
        self
    ):

        for student in self.students:

            yield student

    @log
    def save(
        self,
        filename: str
    ) -> None:

        data = [

            student.to_dict()

            for student
            in self.students
        ]

        with open(
            filename,
            "w",
            encoding="utf-8"
        ) as file:

            json.dump(
                data,
                file,
                ensure_ascii=False,
                indent=2
            )


manager = StudentManager()


manager.add_student(
    Student(
        "홍길동",
        90,
        80,
        70
    )
)

manager.add_student(
    Student(
        "이순신",
        100,
        95,
        90
    )
)


for student in (
    manager.students_generator()
):

    print(
        student.name,
        student.total,
        f"{student.average:.2f}",
        student.grade
    )


manager.save(
    "students.json"
)
```

# 교육 관점에서의 핵심

파이썬을 가르칠 때 개별 문법을 따로 외우게 하기보다 다음과 같이 프로그램이 발전하는 모습을 보여주는 것이 중요하다.

```text
print()

↓
변수

↓
input()

↓
연산자

↓
if

↓
for / while

↓
list / dict

↓
함수

↓
파일 저장

↓
클래스

↓
모듈

↓
예외처리

↓
라이브러리

↓
데코레이터 / 제너레이터

↓
실제 서비스 프로그램
```

결국 파이썬 학습의 핵심은 **문법을 많이 외우는 것이 아니라 데이터가 어떻게 들어와서 → 처리되고 → 저장되고 → 객체와 함수로 구조화되는가를 이해하는 것**이다.
