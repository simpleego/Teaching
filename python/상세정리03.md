### 1. 데이터와 데이터형 그리고 변수
변수는 데이터를 저장하는 그릇이며, 파이썬은 동적 타이핑 언어이므로 변수의 타입을 미리 선언할 필요가 없습니다.

**[예제: 성적표 데이터 저장]**
```python
# 1. 기본 데이터형 (int, float, str, bool, None)
student_name = "홍길동"       # str (문자열)
student_id = 2023001         # int (정수)
math_score = 95.5            # float (실수)
is_passed = True             # bool (불리언)
remarks = None               # NoneType (값이 없음)

# 2. 컬렉션 데이터형 (list, tuple, dict, set)
scores_list = [90, 85, 100]  # list (변경 가능한 순서 있는 나열)
scores_tuple = (90, 85, 100) # tuple (변경 불가능한 순서 있는 나열)
score_dict = {"math": 95, "eng": 88} # dict (키-값 쌍)
unique_scores = {90, 95, 100} # set (중복 없는 집합)

# 3. 타입 확인 및 형변환
print(type(student_name))    # <class 'str'>
print(str(student_id))       # '2023001' (정수 -> 문자열)
print(int(math_score))       # 95 (실수 -> 정수, 버림)
```

### 2. 연산자
데이터를 처리하고 비교하는 다양한 연산자입니다.

**[예제: 계산기 및 조건 판단]**
```python
# 1. 산술 연산자
a, b = 10, 3
print(a + b, a - b, a * b)   # 13 7 30
print(a / b)                 # 3.333... (나눗셈)
print(a // b)                # 3 (몫)
print(a % b)                 # 1 (나머지)
print(a ** b)                # 1000 (거듭제곱)

# 2. 비교 및 논리 연산자
print(a > b)                 # True
print(a == b)                # False
print(a != b)                # True
print(a > 5 and b < 5)       # True (논리곱)
print(a > 15 or b < 5)       # True (논리합)
print(not (a > b))           # False (논리부정)

# 3. 멤버십 및 동일성 연산자
fruits = ["apple", "banana"]
print("apple" in fruits)     # True (멤버십)
print(fruits is not None)    # True (동일성/메모리 주소 비교)

# 4. 대입 연산자 및 왈rus 연산자 (:=)
c = 5
c += 2                       # c = c + 2
# 왈rus 연산자: 할당과 동시에 값 평가
if (n := len(fruits)) > 1:   
    print(f"과일은 {n}개입니다.") # 출력: 과일은 2개입니다.
```

### 3. 제어문
프로그램의 흐름을 제어하는 조건문과 반복문입니다.

**[예제: 자판기 시스템]**
```python
# 1. 조건문 (if-elif-else) 및 match-case (Python 3.10+)
money = 1500
menu = {"cola": 1000, "coffee": 1500, "water": 500}

def select_drink(choice):
    match choice:
        case "cola" | "coffee" | "water":
            return menu[choice]
        case _:
            return 0 # 매칭되는 경우 없음

price = select_drink("coffee")

if money >= price:
    print(f"선택한 음료를 뽑습니다. 잔돈: {money - price}원")
elif money > 0:
    print("돈이 부족합니다.")
else:
    print("돈을 넣어주세요.")

# 2. 반복문 (for, while) 및 제어어 (break, continue)
print("\n--- 자판기 메뉴 순회 ---")
for item, cost in menu.items():
    if cost > money:
        continue # 구매 불가 시 건너뜀
    print(f"{item} 판매중 ({cost}원)")

# 3. 리스트 내포 (List Comprehension) - 제어문의 축약형
available_items = [item for item, cost in menu.items() if cost <= money]
print(f"구매 가능한 메뉴: {available_items}")

# 4. while 루프
balance = 2000
while balance >= 500:
    print("500원짜리 물을 구매합니다.")
    balance -= 500
```

### 4. 함수, 파일 입출력
코드의 재사용성을 높이는 함수와 데이터를 영속화하는 파일 I/O입니다.

**[예제: 성적표 계산 및 파일 저장]**
```python
# 1. 함수 정의 및 다양한 인자
def calculate_grade(name, *scores, **extra_info):
    # *args: 튜플로 가변 인자 받음, **kwargs: 딕셔너리로 가변 키워드 인자 받음
    avg = sum(scores) / len(scores) if scores else 0
    
    # 람다 함수 (익명 함수)
    get_pass_fail = lambda score: "Pass" if score >= 60 else "Fail"
    
    return {
        "name": name,
        "avg": avg,
        "status": get_pass_fail(avg),
        "info": extra_info
    }

result = calculate_grade("김철수", 80, 90, 70, major="Computer Science")

# 2. f-string 포매팅
print(f"{result['name']}의 평균: {result['avg']:.2f}, 상태: {result['status']}")

# 3. 파일 입출력 (with 구문 사용 - 자동 close)
# 쓰기 모드
with open("grade_report.txt", "w", encoding="utf-8") as f:
    f.write(f"이름: {result['name']}\n")
    f.write(f"평균: {result['avg']}\n")

# 읽기 모드
with open("grade_report.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print("\n[파일 내용]")
    print(content)
```

### 5. 클래스, 모듈, 패키지
객체 지향 프로그래밍과 코드 구조화입니다.

**[예제: 자판기 클래스 설계]**
```python
# 1. 클래스 정의, 상속, 캡슐화
class Machine:
    def __init__(self, name):
        self.name = name  # 공개 속성
        
    def turn_on(self):
        print(f"{self.name} 기기가 켜졌습니다.")

class VendingMachine(Machine): # 상속
    def __init__(self, name, initial_balance=0):
        super().__init__(name) # 부모 클래스 초기화
        self.__balance = initial_balance # __를 붙여 비공개(Private) 속성으로 설정
        
    # @property: 메서드를 속성처럼 사용 (Getter)
    @property
    def balance(self):
        return self.__balance
        
    @balance.setter
    def balance(self, amount):
        if amount >= 0:
            self.__balance = amount
        else:
            raise ValueError("잔액은 0 이상이어야 합니다.")

    # 매직 메서드 (객체 출력 형식 정의)
    def __str__(self):
        return f"[{self.name}] 현재 잔액: {self.__balance}원"

vm = VendingMachine("코카콜라 자판기", 5000)
vm.turn_on()
print(vm) # __str__ 호출
vm.balance = 10000 # setter 호출
# vm.__balance = 0  # 주석 해제 시 에러 발생 (캡슐화됨)
```
*모듈과 패키지 개념:*
위 코드를 `vending.py`에 저장하면 **모듈**이 됩니다. 여러 모듈을 `__init__.py`가 있는 폴더(예: `my_system/`)에 넣으면 **패키지**가 됩니다. `from my_system.vending import VendingMachine`로 호출합니다.

### 6. 예외처리
예상치 못한 에러로부터 프로그램을 보호합니다.

**[예제: 계산기 및 자판기 에러 핸들링]**
```python
# 1. 커스텀 예외 클래스 정의
class InsufficientBalanceError(Exception):
    def __init__(self, balance, price):
        super().__init__(f"잔액({balance}원)이 가격({price}원)보다 부족합니다.")
        self.balance = balance
        self.price = price

# 2. try-except-else-finally 및 raise
def safe_divide(a, b):
    try:
        result = a / b
    except ZeroDivisionError as e:
        print(f"0으로 나눌 수 없습니다: {e}")
    except TypeError:
        print("숫자만 입력해주세요.")
    else:
        # 예외가 발생하지 않았을 때 실행
        print(f"계산 성공: {result}")
        return result
    finally:
        # 성공/실패 무관하게 항상 실행 (자원 반납 등)
        print("계산기 로직 종료.\n")

safe_divide(10, 2)
safe_divide(10, 0)

# 3. 커스텀 예외 발생시키기 (raise)
def buy_item(balance, price):
    if balance < price:
        raise InsufficientBalanceError(balance, price)
    return balance - price

try:
    buy_item(500, 1000)
except InsufficientBalanceError as e:
    print(f"구매 실패: {e}")
```

### 8. 클로저, 데코레이터, 제너레이터, 타입 애너테이션
파이썬의 고급 기능으로, 함수형 프로그래밍 패러다임과 코드의 안정성을 높입니다.

**[예제: 고급 로깅 계산기 및 대용량 성적 처리]**
```python
import time
from functools import wraps
from typing import List, Dict, Union, Optional, Generator

# 1. 타입 애너테이션 (Type Annotations)
# 변수와 함수의 입출력 타입을 명시하여 가독성과 IDE 지원을 높임
def calculate_average(scores: List[Union[int, float]]) -> float:
    if not scores:
        return 0.0
    return sum(scores) / len(scores)

# 2. 클로저 (Closure)
# 내부 함수가 외부 함수의 변수를 기억하는 현상
def make_multiplier(n: int):
    def multiplier(x: int) -> int:
        return x * n # 외부 변수 n을 기억
    return multiplier

times_three = make_multiplier(3)
print(f"클로저 테스트: 10 * 3 = {times_three(10)}")

# 3. 데코레이터 (Decorator)
# 기존 함수의 기능을 수정하거나 래핑할 때 사용 (AOP)
def log_execution_time(func):
    @wraps(func) # 원본 함수의 메타데이터 유지
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"[{func.__name__}] 실행 시간: {end - start:.6f}초")
        return result
    return wrapper

@log_execution_time # calculate_sum = log_execution_time(calculate_sum)
def calculate_sum(n: int) -> int:
    return sum(range(n))

calculate_sum(1000000)

# 4. 제너레이터 (Generator)
# yield를 사용하여 데이터를 한 번에 메모리에 올리지 않고 하나씩 생성 (메모리 효율)
def generate_student_records(count: int) -> Generator[Dict[str, str], None, None]:
    for i in range(count):
        # yield는 값을 반환하고 함수 상태를 일시 중지
        yield {"id": i, "name": f"Student_{i}", "status": "Active"}

# 제너레이터 사용 (대용량 데이터 처리 시 유용)
print("\n--- 제너레이터 테스트 ---")
gen = generate_student_records(3)
for record in gen:
    print(record)

# 제너레이터 표현식 (리스트 내포와 유사하지만 () 사용)
squared_gen = (x**2 for x in range(5))
print(f"제너레이터 객체: {squared_gen}")
print(f"다음 값: {next(squared_gen)}")
```
