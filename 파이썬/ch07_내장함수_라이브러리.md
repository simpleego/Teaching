## 7.8 collections

```python
from collections import Counter, defaultdict, OrderedDict, deque, namedtuple, ChainMap

# Counter: 개수 세기
sales = ["콜라", "물", "콜라", "커피", "콜라", "물"]
cnt = Counter(sales)
print(cnt)                         # Counter({'콜라': 3, '물': 2, '커피': 1})
print(cnt.most_common(2))          # [('콜라', 3), ('물', 2)]
print(cnt["주스"])                  # 0 (KeyError 없음)
cnt.update(["커피", "커피"])
print(cnt + Counter(["물"]))

# defaultdict: 키가 없을 때 기본값 자동 생성
by_grade = defaultdict(list)
for name, g in [("홍길동", "B"), ("김영희", "A"), ("박민수", "B")]:
    by_grade[g].append(name)
print(dict(by_grade))              # {'B': ['홍길동', '박민수'], 'A': ['김영희']}

# namedtuple: 이름으로 접근하는 튜플
Student = namedtuple("Student", ["name", "kor", "eng"])
s = Student("홍길동", 90, 85)
print(s.name, s[1], s._asdict())

# deque: 양쪽 끝 삽입/삭제가 빠른 큐
q = deque([1, 2, 3], maxlen=5)
q.appendleft(0); q.append(4)
print(q.popleft(), q.pop(), q)
q.rotate(1)
print(q)

# OrderedDict: 순서 조작 메서드 제공
od = OrderedDict(a=1, b=2, c=3)
od.move_to_end("a")
print(list(od))                    # ['b', 'c', 'a']

# ChainMap: 여러 딕셔너리를 하나처럼 조회 (설정 우선순위)
default = {"theme": "light", "lang": "ko"}
user = {"theme": "dark"}
cfg = ChainMap(user, default)
print(cfg["theme"], cfg["lang"])   # dark ko
```

## 7.9 itertools

```python
import itertools as it

print(list(it.permutations("ABC", 2)))           # 순열
print(list(it.combinations([1, 2, 3, 4], 2)))    # 조합
print(list(it.combinations_with_replacement("AB", 2)))
print(list(it.product("AB", [1, 2])))            # 데카르트 곱
print(list(it.chain([1, 2], [3], (4, 5))))       # 이어붙이기
print(list(it.accumulate([1, 2, 3, 4])))         # 누적합 [1, 3, 6, 10]
print(list(it.islice(it.count(10, 5), 4)))       # [10, 15, 20, 25]
print(list(it.islice(it.cycle("AB"), 5)))        # A B A B A
print(list(it.repeat("x", 3)))
print(list(it.takewhile(lambda x: x < 3, [1, 2, 3, 1])))   # [1, 2]
print(list(it.dropwhile(lambda x: x < 3, [1, 2, 3, 1])))   # [3, 1]
print(list(it.zip_longest("AB", [1, 2, 3], fillvalue="-")))
print(list(it.pairwise([1, 2, 3, 4])))           # (1,2),(2,3),(3,4)  3.10+

# groupby: '연속된' 같은 키끼리 묶음 → 보통 먼저 정렬
data = sorted([("B", "홍길동"), ("A", "김영희"), ("B", "박민수")])
for grade, group in it.groupby(data, key=lambda x: x[0]):
    print(grade, [name for _, name in group])
```

## 7.10 functools / operator

```python
from functools import reduce, partial, lru_cache, cache, wraps, total_ordering
from operator import add, mul, itemgetter, attrgetter

print(reduce(add, [1, 2, 3, 4]))          # 10  ((1+2)+3)+4
print(reduce(mul, range(1, 6)))           # 120

# partial: 인자 일부를 고정한 새 함수
to_int_from_bin = partial(int, base=2)
print(to_int_from_bin("1010"))            # 10

# lru_cache: 결과 캐싱(메모이제이션) → 8장 데코레이터 참고
@lru_cache(maxsize=None)
def fib(n):
    return n if n < 2 else fib(n - 1) + fib(n - 2)
print(fib(100))
print(fib.cache_info())

# itemgetter / attrgetter: 정렬 key에 유용
rows = [("홍길동", 85), ("김영희", 95)]
print(sorted(rows, key=itemgetter(1)))

# total_ordering: __eq__와 하나의 비교만 정의하면 나머지 자동 생성
@total_ordering
class Ver:
    def __init__(self, n): self.n = n
    def __eq__(self, o): return self.n == o.n
    def __lt__(self, o): return self.n < o.n
print(Ver(1) >= Ver(2))                   # False
```

## 7.12 re (정규 표현식)

```python
import re

text = "홍길동 010-1234-5678, 김영희 010-9876-5432, 이메일 hong@test.com"

print(re.findall(r"\d{3}-\d{4}-\d{4}", text))       # 모든 전화번호
m = re.search(r"(\w+)@(\w+)\.com", text)              # 첫 매치
print(m.group(), m.group(1), m.groups(), m.span())
print(re.match(r"홍", text) is not None)             # 문자열 '시작'에서만 검사
print(re.fullmatch(r"\d{4}", "2026") is not None)    # 전체 일치
print(re.sub(r"\d{4}-(\d{4})", r"****-\1", text))   # 치환(마스킹)
print(re.split(r"[,\s]+", "a, b  c,d"))

# 이름 있는 그룹 + 컴파일
pattern = re.compile(r"(?P<name>\w+) (?P<phone>010-\d{4}-\d{4})")
for mt in pattern.finditer(text):
    print(mt["name"], mt["phone"])

# 플래그
print(re.findall(r"python", "Python PYTHON", flags=re.IGNORECASE))
```

## 🧪 예제 프로그램 ①: 성적표 v7 – 표준 라이브러리 총동원

```python
# ===== 성적표 v7 =====
import random, statistics as st, bisect
from collections import namedtuple, defaultdict, Counter
from datetime import datetime
from operator import attrgetter

Student = namedtuple("Student", "name cls kor eng math")
SUBJECTS = ("kor", "eng", "math")
CUTS, GRADES = [60, 70, 80, 90], "FDCBA"

random.seed(7)
NAMES = ["홍길동", "김영희", "이철수", "박민수", "최지우", "정다은", "오세훈", "한가람"]
students = [Student(n, random.choice(["1반", "2반"]),
                    *(random.randint(40, 100) for _ in SUBJECTS)) for n in NAMES]

def avg(s): return st.mean(getattr(s, sub) for sub in SUBJECTS)
def grade(a): return GRADES[bisect.bisect(CUTS, a)]

print(f"📋 성적 리포트 ({datetime.now():%Y-%m-%d %H:%M})")
print(f"{'이름':<5}{'반':<4}{'국':>4}{'영':>4}{'수':>4}{'평균':>7}{'학점':>3}")
for s in sorted(students, key=avg, reverse=True):
    a = avg(s)
    print(f"{s.name:<5}{s.cls:<4}{s.kor:>5}{s.eng:>5}{s.math:>5}{a:>8.1f}{grade(a):>4}")

# 반별 평균 (defaultdict)
by_class = defaultdict(list)
for s in students:
    by_class[s.cls].append(avg(s))
for c, avgs in sorted(by_class.items()):
    print(f"{c} 평균 {st.mean(avgs):.1f} (표준편차 {st.pstdev(avgs):.1f})")

# 학점 분포 (Counter)
dist = Counter(grade(avg(s)) for s in students)
for g in "ABCDF":
    print(f"{g}: {'■' * dist[g]} {dist[g]}")

# 과목별 최고점 학생 (max + attrgetter)
for sub in SUBJECTS:
    top = max(students, key=attrgetter(sub))
    print(f"{sub} 최고점: {top.name} ({getattr(top, sub)})")
```

## 🧪 예제 프로그램 ②: 자판기 판매 리포트

```python
# ===== 자판기 판매 로그 분석 =====
import random
from collections import Counter
from datetime import datetime, timedelta
from itertools import groupby

PRICES = {"콜라": 1500, "사이다": 1300, "커피": 1000, "물": 800}
random.seed(1)
start = datetime(2026, 10, 1, 9)
logs = [(start + timedelta(minutes=random.randint(0, 60 * 24 * 3)),
         random.choices(list(PRICES), weights=[4, 2, 3, 1])[0]) for _ in range(40)]
logs.sort()

cnt = Counter(item for _, item in logs)
print("🏆 판매 순위:", cnt.most_common())
print("💰 총 매출:", f"{sum(PRICES[i] * n for i, n in cnt.items()):,}원")

print("📅 일별 매출")
for day, group in groupby(logs, key=lambda x: x[0].date()):
    items = [i for _, i in group]
    print(f"  {day:%m/%d}: {len(items)}건 {sum(PRICES[i] for i in items):,}원")
```
