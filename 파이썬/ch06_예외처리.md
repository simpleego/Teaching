## 6.1 오류의 종류

```python
# ① 문법 오류(SyntaxError): 실행 전에 발견 → 예외처리로 잡을 수 없음
# print("hello"     ← 괄호 누락

# ② 예외(Exception): 실행 중 발생하는 오류
# print(10 / 0)          # ZeroDivisionError
# int("abc")             # ValueError
# [1, 2][5]              # IndexError
# {"a": 1}["b"]          # KeyError
# "1" + 1                # TypeError
# undefined_var          # NameError
# open("없는파일.txt")    # FileNotFoundError
# None.upper()           # AttributeError
# import 없는모듈          # ModuleNotFoundError
```

### 주요 내장 예외 계층 (일부)
```
BaseException
 ├── SystemExit, KeyboardInterrupt
 └── Exception
      ├── ArithmeticError
      │    └── ZeroDivisionError, OverflowError
      ├── LookupError
      │    └── IndexError, KeyError
      ├── ValueError
      ├── TypeError
      ├── NameError
      ├── AttributeError
      ├── OSError
      │    └── FileNotFoundError, PermissionError
      └── ImportError
           └── ModuleNotFoundError
```

## 6.2 try - except

```python
# ① 기본형
try:
    result = 10 / 0
except ZeroDivisionError:
    print("0으로 나눌 수 없습니다.")

# ② 예외 객체 받기 (as)
try:
    int("abc")
except ValueError as e:
    print("오류 내용:", e)          # invalid literal for int() ...
    print("오류 타입:", type(e).__name__)

# ③ 여러 except (위에서부터 검사 → 구체적인 예외를 먼저!)
def safe_get(data, idx):
    try:
        return 100 / data[idx]
    except IndexError:
        return "인덱스 범위 초과"
    except ZeroDivisionError:
        return "0으로 나눔"
    except TypeError as e:
        return f"타입 오류: {e}"

print(safe_get([1, 0], 5))
print(safe_get([1, 0], 1))
print(safe_get([1, "a"], 1))

# ④ 여러 예외를 튜플로 한 번에
try:
    {"a": 1}["b"]
except (KeyError, IndexError) as e:
    print("조회 실패:", repr(e))

# ⑤ 모든 예외 (Exception) – 마지막 안전망으로만 사용
try:
    None.upper()
except Exception as e:
    print(f"예상 못 한 오류: {e!r}")

# ⚠️ bare except(except:)는 Ctrl+C까지 막으므로 사용하지 마세요.
```

## 6.3 else와 finally

```python
def read_number(text):
    try:
        num = int(text)
    except ValueError:
        print(f"  '{text}'는 숫자가 아닙니다.")
        return None
    else:
        # 예외가 '발생하지 않았을 때만' 실행
        print(f"  변환 성공: {num}")
        return num
    finally:
        # 예외 여부와 관계없이 '항상' 실행 (return 직전에도 실행됨)
        print("  [finally] 입력 처리 종료")

read_number("42")
read_number("4x")

# finally의 전형적인 용도: 자원 정리
f = None
try:
    f = open("memo.txt", "w", encoding="utf-8")
    f.write("내용")
finally:
    if f:
        f.close()
        print("파일을 닫았습니다.")
```

**실행 흐름 정리**

| 상황 | try | except | else | finally |
|---|---|---|---|---|
| 예외 없음 | ✔ | ✘ | ✔ | ✔ |
| 예외 발생(처리됨) | 중단 | ✔ | ✘ | ✔ |
| 예외 발생(처리 안 됨) | 중단 | ✘ | ✘ | ✔ 후 예외 전파 |

## 6.4 raise – 예외 직접 발생시키기

```python
def set_score(score):
    if not isinstance(score, (int, float)):
        raise TypeError("점수는 숫자여야 합니다")
    if not 0 <= score <= 100:
        raise ValueError(f"점수 범위 오류: {score}")
    return score

for s in (90, 120, "A"):
    try:
        print("저장:", set_score(s))
    except (ValueError, TypeError) as e:
        print(f"{type(e).__name__}: {e}")

# 예외 다시 던지기 (로그만 남기고 상위로 전파)
def process():
    try:
        set_score(-5)
    except ValueError:
        print("process에서 기록 후 다시 전달")
        raise                     # 같은 예외를 그대로 다시 발생

try:
    process()
except ValueError as e:
    print("최상위에서 처리:", e)
```

## 6.5 예외 체이닝 (raise ... from ...)

```python
class ConfigError(Exception):
    pass

def load_price(text):
    try:
        return int(text)
    except ValueError as e:
        # 원인 예외(e)를 연결하여 새 예외로 변환
        raise ConfigError(f"가격 설정 오류: {text!r}") from e

try:
    load_price("천원")
except ConfigError as e:
    print(e)
    print("원인:", repr(e.__cause__))

# from None: 원인 예외를 숨김
def get_item(d, key):
    try:
        return d[key]
    except KeyError:
        raise LookupError(f"'{key}' 상품 없음") from None
```

## 6.6 사용자 정의 예외

```python
# Exception을 상속하여 만든다. 계층 구조로 설계하면 묶어서 처리하기 좋다.
class VendingError(Exception):
    """자판기 관련 최상위 예외"""

class SoldOutError(VendingError):
    def __init__(self, item):
        self.item = item
        super().__init__(f"{item} 품절")

class InsufficientFundsError(VendingError):
    def __init__(self, need, have):
        self.need, self.have = need, have
        super().__init__(f"잔액 부족: {need - have}원 더 필요")

try:
    raise InsufficientFundsError(1500, 1000)
except SoldOutError as e:
    print("품절 처리:", e.item)
except VendingError as e:          # 하위 예외들을 한 번에 잡음
    print("자판기 오류:", e, "| 부족액:", e.need - e.have)
```

## 6.7 assert – 개발 중 조건 검사

```python
def average(scores):
    assert len(scores) > 0, "점수 리스트가 비어 있습니다"   # 조건이 False면 AssertionError
    return sum(scores) / len(scores)

print(average([90, 80]))
try:
    average([])
except AssertionError as e:
    print("AssertionError:", e)

# ⚠️ python -O 옵션으로 실행하면 assert는 무시됨
#    → 사용자 입력 검증에는 raise를 사용하고, assert는 '개발자의 가정' 확인용
```

## 6.8 컨텍스트 매니저와 with

```python
# with 문은 내부적으로 __enter__ / __exit__ 를 호출 → 예외가 나도 정리 보장

class Timer:
    import time as _t
    def __enter__(self):
        self.start = self._t.perf_counter()
        print("▶ 시작")
        return self                       # as 뒤 변수로 전달됨
    def __exit__(self, exc_type, exc_val, tb):
        elapsed = self._t.perf_counter() - self.start
        print(f"■ 종료 ({elapsed:.4f}초)")
        if exc_type is ZeroDivisionError:
            print("  0 나눗셈 예외를 처리했습니다.")
            return True                   # True 반환 → 예외를 삼킴(전파 안 함)
        return False                      # 그 외 예외는 전파

with Timer():
    sum(range(1_000_000))

with Timer():
    1 / 0                                 # 예외가 나도 __exit__ 실행
print("프로그램 계속 진행")

# contextlib로 간단히 만들기 (제너레이터 기반, 8장 참고)
from contextlib import contextmanager, suppress

@contextmanager
def transaction(name):
    print(f"[{name}] 트랜잭션 시작")
    try:
        yield
        print(f"[{name}] 커밋")
    except Exception as e:
        print(f"[{name}] 롤백: {e}")

with transaction("구매"):
    raise ValueError("재고 없음")

# suppress: 특정 예외 무시
with suppress(FileNotFoundError):
    open("없는파일.txt")
print("무시하고 진행")
```

## 6.10 ExceptionGroup과 except* (파이썬 3.11+)

```python
# 여러 예외를 한 번에 묶어 던지고 종류별로 처리 (병렬 작업 결과 처리 등)
errors = []
for value in ["10", "x", "0"]:
    try:
        print(100 / int(value))
    except Exception as e:
        errors.append(e)

try:
    if errors:
        raise ExceptionGroup("입력 처리 오류들", errors)
except* ValueError as eg:
    print("ValueError들:", [str(e) for e in eg.exceptions])
except* ZeroDivisionError as eg:
    print("ZeroDivisionError들:", len(eg.exceptions), "건")

# add_note: 예외에 부가 정보 추가 (3.11+)
try:
    try:
        int("abc")
    except ValueError as e:
        e.add_note("성적 CSV 3행 국어 점수")
        raise
except ValueError as e:
    print(e, e.__notes__)
```

## 🧪 예제 프로그램 ①: 계산기 v4 – 견고한 계산기

```python
# ===== 계산기 v4: 예외처리 적용 =====
import operator

class CalculatorError(Exception):
    """계산기 예외"""

OPS = {"+": operator.add, "-": operator.sub, "*": operator.mul,
       "/": operator.truediv, "//": operator.floordiv, "%": operator.mod, "**": operator.pow}

def parse(expr):
    try:
        a, op, b = expr.split()
    except ValueError:
        raise CalculatorError("형식은 '숫자 연산자 숫자'입니다") from None
    try:
        return float(a), op, float(b)
    except ValueError as e:
        raise CalculatorError(f"숫자가 아닌 값: {expr}") from e

def calculate(expr):
    a, op, b = parse(expr)
    if op not in OPS:
        raise CalculatorError(f"지원하지 않는 연산자 '{op}'")
    try:
        return OPS[op](a, b)
    except ZeroDivisionError:
        raise CalculatorError("0으로 나눌 수 없습니다") from None
    except OverflowError:
        raise CalculatorError("결과가 너무 큽니다") from None

tests = ["3 + 4", "10 / 0", "2 ** 10000", "a * 3", "5 ^ 2", "7", "9 // 2"]
success = fail = 0
for t in tests:
    try:
        result = calculate(t)
    except CalculatorError as e:
        fail += 1
        print(f"✗ {t:<12} → {e}")
    else:
        success += 1
        print(f"✔ {t:<12} = {result:g}")
    finally:
        pass
print(f"성공 {success}건 / 실패 {fail}건")
```

## 🧪 예제 프로그램 ②: 자판기 v4 – 사용자 정의 예외

```python
# ===== 자판기 v4 =====
class VendingError(Exception): pass
class InvalidCoinError(VendingError): pass
class SoldOutError(VendingError): pass
class InsufficientFundsError(VendingError): pass
class UnknownItemError(VendingError): pass

class VendingMachine:
    COINS = {50, 100, 500, 1000}

    def __init__(self, items):
        self.items = items          # {"코드": [이름, 가격, 재고]}
        self.balance = 0

    def insert(self, coin):
        if coin not in self.COINS:
            raise InvalidCoinError(f"{coin}원 동전/지폐는 사용 불가")
        self.balance += coin

    def buy(self, code):
        if code not in self.items:
            raise UnknownItemError(f"상품 번호 {code} 없음")
        name, price, stock = self.items[code]
        if stock <= 0:
            raise SoldOutError(f"{name} 품절")
        if self.balance < price:
            raise InsufficientFundsError(f"{name}: {price - self.balance}원 부족")
        self.items[code][2] -= 1
        self.balance -= price
        return name

vm = VendingMachine({"1": ["콜라", 1500, 1], "2": ["물", 800, 3]})
actions = [("insert", 1000), ("insert", 10), ("buy", "1"), ("insert", 1000),
           ("buy", "1"), ("buy", "1"), ("buy", "9"), ("buy", "2")]

for action, arg in actions:
    try:
        if action == "insert":
            vm.insert(arg)
            print(f"＋{arg}원 → 잔액 {vm.balance}")
        else:
            print(f"✔ {vm.buy(arg)} 구매 → 잔액 {vm.balance}")
    except InsufficientFundsError as e:
        print("💰", e)
    except VendingError as e:
        print("⚠️", type(e).__name__, "-", e)
```

## 🧪 예제 프로그램 ③: 성적표 v6 – CSV 입력 검증

```python
# ===== 성적표 v6 =====
import csv, io

# 실제로는 파일을 열지만, 여기서는 문자열로 CSV를 흉내냄
raw = """이름,국어,영어,수학
홍길동,90,85,77
김영희,95,abc,88
이철수,55,38
박민수,72,181,64
"""

class ScoreFormatError(ValueError):
    def __init__(self, line, msg):
        super().__init__(f"{line}행: {msg}")

def parse_row(line_no, row):
    if len(row) != 4:
        raise ScoreFormatError(line_no, f"열 개수 {len(row)}개 (4개 필요)")
    name, *scores = row
    try:
        scores = [int(s) for s in scores]
    except ValueError as e:
        raise ScoreFormatError(line_no, "숫자가 아닌 점수") from e
    for s in scores:
        if not 0 <= s <= 100:
            raise ScoreFormatError(line_no, f"범위를 벗어난 점수 {s}")
    return name, scores

valid, errors = [], []
reader = csv.reader(io.StringIO(raw))
next(reader)                                   # 헤더 건너뛰기
for line_no, row in enumerate(reader, start=2):
    try:
        valid.append(parse_row(line_no, row))
    except ScoreFormatError as e:
        errors.append(str(e))

print("✔ 정상 데이터:")
for name, scores in valid:
    print(f"  {name}: {scores} 평균 {sum(scores)/3:.1f}")
print("✗ 오류 목록:")
for err in errors:
    print("  -", err)
```
