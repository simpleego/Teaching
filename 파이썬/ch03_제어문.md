## 3.1 들여쓰기(Indentation)

파이썬은 중괄호 `{}` 대신 **들여쓰기(공백 4칸 권장)**로 코드 블록을 구분합니다. 블록 시작 줄은 반드시 콜론(`:`)으로 끝납니다.

```python
score = 90
if score >= 60:
    print("합격")          # if 블록 안 (4칸 들여쓰기)
    print("축하합니다")      # 같은 블록
print("프로그램 종료")      # 블록 밖
```

## 3.2 if / elif / else

```python
# ① 단순 if
age = 20
if age >= 19:
    print("성인입니다")

# ② if - else
temperature = 15
if temperature >= 20:
    print("반팔")
else:
    print("긴팔")

# ③ if - elif - else : 학점 계산
score = 83
if score >= 90:
    grade = "A"
elif score >= 80:          # 90 미만이면서 80 이상
    grade = "B"
elif score >= 70:
    grade = "C"
elif score >= 60:
    grade = "D"
else:
    grade = "F"
print(f"{score}점 → {grade}")

# ④ 중첩 if
is_member = True
amount = 50_000
if amount >= 30_000:
    if is_member:
        print("회원 10% 할인")
    else:
        print("비회원 5% 할인")

# ⑤ 조건 표현식(한 줄 if)
result = "짝수" if amount % 2 == 0 else "홀수"

# ⑥ pass: 아무것도 하지 않음 (나중에 채울 자리)
if amount < 0:
    pass   # TODO: 오류 처리

# ⑦ Truthy/Falsy 활용
cart = []
if not cart:
    print("장바구니가 비어 있습니다")
```

## 3.3 match-case (구조적 패턴 매칭, 파이썬 3.10+)

```python
# ① 값 매칭 (다른 언어의 switch와 비슷)
op = "+"
match op:
    case "+":
        print("덧셈")
    case "-":
        print("뺄셈")
    case "*" | "x":          # | : 여러 패턴 중 하나
        print("곱셈")
    case _:                  # _ : 그 밖의 모든 경우 (default)
        print("알 수 없는 연산자")

# ② 가드(guard) 조건: case 패턴 if 조건
score = 95
match score:
    case s if s >= 90:
        print("A")
    case s if s >= 80:
        print("B")
    case _:
        print("그 외")

# ③ 시퀀스 패턴: 구조 분해
command = ["buy", "콜라", 2]
match command:
    case ["buy", item, qty]:
        print(f"{item} {qty}개 구매")
    case ["refund"]:
        print("환불")
    case ["add", *items]:    # 나머지 전부
        print("여러 개 추가:", items)

# ④ 딕셔너리 패턴
event = {"type": "coin", "value": 500}
match event:
    case {"type": "coin", "value": v}:
        print(f"{v}원 투입")
    case {"type": "select", "item": name}:
        print(f"{name} 선택")
```

## 3.4 for 반복문

```python
# ① 리스트 순회
subjects = ["국어", "영어", "수학"]
for subject in subjects:
    print(subject)

# ② 문자열 순회
for ch in "ABC":
    print(ch, end=" ")     # end: 줄바꿈 대신 출력할 문자
print()

# ③ range(start, stop, step) : stop은 포함하지 않음
for i in range(5):           # 0~4
    print(i, end=" ")
print()
for i in range(1, 11, 2):    # 1, 3, 5, 7, 9
    print(i, end=" ")
print()
for i in range(10, 0, -3):   # 10, 7, 4, 1 (역순)
    print(i, end=" ")
print()

# ④ enumerate: 인덱스와 값을 함께
for idx, subject in enumerate(subjects, start=1):
    print(f"{idx}. {subject}")

# ⑤ zip: 여러 시퀀스를 묶어서 순회
scores = [90, 85, 77]
for subject, score in zip(subjects, scores):
    print(f"{subject}: {score}")

# ⑥ 딕셔너리 순회
stock = {"콜라": 3, "사이다": 5}
for name in stock:                    # 키만
    print(name)
for name, qty in stock.items():       # 키, 값
    print(f"{name} 재고 {qty}")

# ⑦ 튜플 리스트 언패킹 순회
students = [("홍길동", 90), ("김영희", 85)]
for name, score in students:
    print(name, score)

# ⑧ 중첩 for: 구구단
for dan in range(2, 4):
    for i in range(1, 10):
        print(f"{dan}x{i}={dan*i:2}", end="  ")
    print()

# ⑨ 역순·정렬 순회
for s in reversed(subjects):
    print(s)
for s in sorted(scores, reverse=True):
    print(s)
```

## 3.5 while 반복문

```python
# ① 기본 while: 조건이 참인 동안 반복
count = 1
while count <= 5:
    print(count)
    count += 1              # ⚠️ 빠뜨리면 무한 루프!

# ② 누적 합계
total, n = 0, 1
while n <= 100:
    total += n
    n += 1
print("1~100 합:", total)

# ③ 무한 루프 + break (입력 처리에 자주 사용)
balance = 3000
while True:
    if balance < 1000:
        print("잔액 부족, 종료")
        break
    balance -= 1000
    print(f"1000원 사용, 남은 잔액 {balance}")
```

## 3.6 break, continue, else

```python
# break: 반복문 즉시 탈출
for n in range(1, 10):
    if n == 5:
        break
    print(n, end=" ")      # 1 2 3 4
print()

# continue: 이번 회차만 건너뜀
for n in range(1, 10):
    if n % 2 == 0:
        continue
    print(n, end=" ")      # 1 3 5 7 9
print()

# for-else / while-else : break 없이 '정상 종료'했을 때만 else 실행
target = 7
for n in [1, 3, 5]:
    if n == target:
        print("찾음")
        break
else:
    print("못 찾음")        # 이 줄이 실행됨

# 활용: 소수 판별
num = 29
for d in range(2, int(num ** 0.5) + 1):
    if num % d == 0:
        print(f"{num}은 소수가 아님")
        break
else:
    print(f"{num}은 소수")

# 중첩 반복 탈출: 플래그 사용
found = False
for i in range(3):
    for j in range(3):
        if i * j == 2:
            found = True
            break
    if found:
        break
print("i, j =", i, j)
```

## 3.7 컴프리헨션(Comprehension)

```python
# ① 리스트 컴프리헨션: [표현식 for 변수 in 반복가능 if 조건]
squares = [x ** 2 for x in range(1, 6)]
print(squares)                       # [1, 4, 9, 16, 25]

evens = [x for x in range(10) if x % 2 == 0]
print(evens)                         # [0, 2, 4, 6, 8]

# 조건 표현식 포함 (if-else는 for 앞에)
labels = ["Pass" if s >= 60 else "Fail" for s in [90, 40, 70]]
print(labels)

# 중첩 for
pairs = [(i, j) for i in range(2) for j in range(3)]
print(pairs)

# 2차원 리스트 평탄화
matrix = [[1, 2], [3, 4], [5, 6]]
flat = [n for row in matrix for n in row]
print(flat)                          # [1, 2, 3, 4, 5, 6]

# ② 딕셔너리 컴프리헨션
names = ["홍길동", "김영희"]
scores = [90, 85]
score_map = {n: s for n, s in zip(names, scores)}
print(score_map)

# 키·값 뒤집기
inv = {v: k for k, v in score_map.items()}
print(inv)

# ③ 집합 컴프리헨션
lengths = {len(w) for w in ["a", "bb", "cc", "ddd"]}
print(lengths)                       # {1, 2, 3}

# ④ 제너레이터 표현식 (8장에서 자세히) – 괄호 사용, 메모리 절약
total = sum(x ** 2 for x in range(1, 101))
print(total)
```

## 🧪 예제 프로그램 ①: 성적표 v3 – 학급 전체 학점 처리

```python
# ===== 성적표 v3: 제어문 적용 =====
SUBJECTS = ("국어", "영어", "수학")
students = [
    {"name": "홍길동", "scores": [90, 85, 77]},
    {"name": "김영희", "scores": [95, 92, 88]},
    {"name": "이철수", "scores": [55, 38, 70]},
    {"name": "박민수", "scores": [72, 81, 64]},
]

print(f"{'이름':<5}" + "".join(f"{s:>5}" for s in SUBJECTS) + f"{'평균':>7}{'학점':>4}{'비고':>6}")
print("-" * 45)

grade_count = {}                              # 학점별 인원수

for st in students:
    scores = st["scores"]
    avg = sum(scores) / len(scores)

    # 학점 결정 (match-case + 가드)
    match avg:
        case a if a >= 90: grade = "A"
        case a if a >= 80: grade = "B"
        case a if a >= 70: grade = "C"
        case a if a >= 60: grade = "D"
        case _:            grade = "F"

    # 과락(40점 미만) 과목 찾기
    fails = [sub for sub, sc in zip(SUBJECTS, scores) if sc < 40]
    note = "과락" if fails else ""

    grade_count[grade] = grade_count.get(grade, 0) + 1

    line = f"{st['name']:<5}" + "".join(f"{sc:>6}" for sc in scores)
    print(line + f"{avg:>8.1f}{grade:>5}{note:>6}")

print("-" * 45)
for g in sorted(grade_count):
    print(f"{g}학점: {grade_count[g]}명")

# 최고 평균 학생 찾기
best = None
for st in students:
    avg = sum(st["scores"]) / 3
    if best is None or avg > best[1]:
        best = (st["name"], avg)
print(f"최고 평균: {best[0]} ({best[1]:.1f})")
```

## 🧪 예제 프로그램 ②: 계산기 v2 – 메뉴 반복형

```python
# ===== 계산기 v2: while + match =====
# 실제 실습에서는 input()으로 받습니다. 여기서는 입력을 리스트로 대체합니다.
inputs = [("10", "+", "5"), ("8", "/", "0"), ("2", "**", "8"), ("q", "", "")]

for a_str, op, b_str in inputs:          # 실습: while True: a_str = input(...)
    if a_str == "q":
        print("계산기를 종료합니다.")
        break
    if not (a_str.lstrip("-").isdigit() and b_str.lstrip("-").isdigit()):
        print("숫자를 입력하세요.")
        continue

    a, b = int(a_str), int(b_str)
    match op:
        case "+":  result = a + b
        case "-":  result = a - b
        case "*":  result = a * b
        case "/" if b == 0:
            print("0으로 나눌 수 없습니다.")
            continue
        case "/":  result = a / b
        case "**": result = a ** b
        case _:
            print("지원하지 않는 연산자")
            continue
    print(f"{a} {op} {b} = {result}")
```

## 🧪 예제 프로그램 ③: 자판기 v2

```python
# ===== 자판기 v2 =====
menu = {"1": ("콜라", 1500, 2), "2": ("사이다", 1300, 0), "3": ("커피", 1000, 5)}
balance = 0
coins_to_insert = [1000, 500, 300]       # 300원은 잘못된 동전
selections = ["2", "1", "1", "1"]        # 사용자 선택 시뮬레이션

# 동전 투입
for coin in coins_to_insert:
    if coin not in (50, 100, 500, 1000):
        print(f"{coin}원은 사용할 수 없습니다.")
        continue
    balance += coin
print(f"투입 금액: {balance}원")

# 메뉴 출력
for key, (name, price, qty) in menu.items():
    status = "품절" if qty == 0 else f"{price}원"
    print(f"[{key}] {name:<4} {status}")

# 상품 선택
for sel in selections:
    name, price, qty = menu[sel]
    if qty == 0:
        print(f"{name}: 품절입니다.")
    elif balance < price:
        print(f"{name}: 잔액이 부족합니다. (잔액 {balance}원)")
        break
    else:
        balance -= price
        menu[sel] = (name, price, qty - 1)
        print(f"{name} 나왔습니다. 잔액 {balance}원")

print(f"거스름돈: {balance}원")
```
