# Part A. 클래스

## 5.1 클래스와 객체

- **클래스(class)**: 객체를 만들기 위한 설계도 (붕어빵 틀)
- **객체(object) / 인스턴스(instance)**: 클래스로 만든 실체 (붕어빵)
- **속성(attribute)**: 객체가 가진 데이터 / **메서드(method)**: 객체가 가진 함수

```python
class Student:
    """학생을 표현하는 클래스"""

    # 생성자: 객체가 만들어질 때 자동 호출
    def __init__(self, name, kor, eng):
        # self = 지금 만들어지는 객체 자기 자신
        self.name = name          # 인스턴스 변수
        self.kor = kor
        self.eng = eng

    # 인스턴스 메서드: 첫 매개변수는 항상 self
    def total(self):
        return self.kor + self.eng

    def introduce(self):
        print(f"저는 {self.name}이고 총점은 {self.total()}점입니다.")

# 객체 생성
s1 = Student("홍길동", 90, 85)
s2 = Student("김영희", 95, 92)

s1.introduce()
s2.introduce()
print(s1.name, s2.total())

# 속성은 실행 중에 추가/변경 가능
s1.kor = 100
s1.memo = "반장"           # 해당 객체에만 새 속성 추가
print(s1.total(), s1.memo)

print(type(s1))                    # <class '__main__.Student'>
print(isinstance(s1, Student))     # True
print(s1.__dict__)                 # 객체의 속성 딕셔너리
```

## 5.2 클래스 변수 vs 인스턴스 변수

```python
class VendingMachine:
    brand = "PyVend"             # 클래스 변수: 모든 객체가 공유
    machine_count = 0

v2 = VendingMachine("2층")

v1.brand = "Custom"
```

## 5.3 메서드의 종류: 인스턴스 / 클래스 / 정적

```python
class Calculator:
    history_count = 0

    def __init__(self, name):
        self.name = name

    # ① 인스턴스 메서드: 객체의 데이터(self) 사용
    def add(self, a, b):
        Calculator.history_count += 1
        return a + b

    # ② 클래스 메서드: 클래스 자체(cls)를 받음 → 대체 생성자에 자주 사용
    @classmethod
    def from_string(cls, text):
        return cls(text.strip().upper())

    @classmethod
    def get_count(cls):
        return cls.history_count

    # ③ 정적 메서드: self도 cls도 받지 않음 → 클래스와 관련된 유틸 함수
    @staticmethod
    def is_number(value):
        return isinstance(value, (int, float))

calc = Calculator.from_string("  casio ")
print(calc.name)                      # CASIO
print(calc.add(2, 3))
print(Calculator.get_count())         # 1
print(Calculator.is_number("3"))      # False
```

## 5.4 캡슐화: 접근 제어와 프로퍼티

```python
class Account:
    def __init__(self, owner, balance=0):
        self.owner = owner           # public
        self._grade = "일반"          # _protected: 관례상 '내부용'
        self.__balance = balance     # __private: 이름 맹글링(_Account__balance)

    # getter
    @property
    def balance(self):
        return self.__balance

    # setter: 값 검증 로직을 넣을 수 있음
    @balance.setter
    def balance(self, value):
        if value < 0:
            raise ValueError("잔액은 음수가 될 수 없습니다")
        self.__balance = value

    # 읽기 전용 계산 속성
    @property
    def is_vip(self):
        return self.__balance >= 1_000_000

acc = Account("홍길동", 5000)
print(acc.balance)       # 메서드가 아닌 속성처럼 접근 → 5000
acc.balance = 2_000_000  # setter 호출
print(acc.is_vip)        # True
# acc.is_vip = False     # AttributeError (setter 없음)
# print(acc.__balance)   # AttributeError
print(acc._Account__balance)   # 맹글링된 이름 (권장하지 않음)
```

## 5.5 상속(Inheritance)과 오버라이딩

```python
class Product:
    def __init__(self, name, price):
        self.name = name
        self.price = price

    def info(self):
        return f"{self.name}: {self.price}원"

# 자식 클래스(Product를 상속)
class Drink(Product):
    def __init__(self, name, price, ml):
        super().__init__(name, price)    # 부모 생성자 호출
        self.ml = ml                     # 자식만의 속성 추가

    # 메서드 오버라이딩(재정의)
    def info(self):
        base = super().info()            # 부모 메서드 결과 재사용
        return f"{base} ({self.ml}ml)"

class Snack(Product):
    def info(self):
        return f"[과자] {super().info()}"

d = Drink("콜라", 1500, 500)
s = Snack("감자칩", 2000)
print(d.info())
print(s.info())

print(issubclass(Drink, Product))   # True
print(isinstance(d, Product))       # True
print(Drink.__mro__)                # 메서드 탐색 순서

# 다중 상속
class Cold:
    def temp(self): return "차가움"
class IcedCoffee(Cold, Drink):      # 왼쪽 부모 우선 (MRO)
    pass
ic = IcedCoffee("아이스커피", 1800, 350)
print(ic.temp(), ic.info())
```

## 5.6 다형성과 추상 클래스

```python
from abc import ABC, abstractmethod

class Payment(ABC):                 # 추상 클래스: 직접 객체 생성 불가
    @abstractmethod
    def pay(self, amount):          # 자식 클래스가 반드시 구현해야 함
        pass

class Cash(Payment):
    def pay(self, amount):
        return f"현금 {amount}원 결제"

class Card(Payment):
    def pay(self, amount):
        return f"카드 {amount}원 결제 (수수료 포함 {amount * 1.02:.0f}원)"

# 다형성: 같은 메시지(pay)에 객체마다 다르게 반응
for method in [Cash(), Card()]:
    print(method.pay(1500))

# Payment()   # TypeError: 추상 클래스는 인스턴스화 불가

# 덕 타이핑: 상속 관계 없어도 같은 메서드만 있으면 동작
class Coupon:
    def pay(self, amount):
        return f"쿠폰으로 {amount}원 결제"
print(Coupon().pay(1000))
```

## 5.7 매직 메서드(특별 메서드, dunder)

```python
class Score:
    def __init__(self, subject, value):
        self.subject = subject
        self.value = value

    def __str__(self):                 # print(), str() 시 (사용자용)
        return f"{self.subject} {self.value}점"

    def __repr__(self):                # 개발자용 표현 (리스트 출력 등)
        return f"Score({self.subject!r}, {self.value})"

    def __add__(self, other):          # + 연산자
        return self.value + (other.value if isinstance(other, Score) else other)

    def __radd__(self, other):         # sum() 지원 (0 + Score)
        return self.__add__(other)

    def __eq__(self, other):           # ==
        return self.value == other.value

    def __lt__(self, other):           # <  (정렬 가능해짐)
        return self.value < other.value

    def __bool__(self):                # bool() / if 판정
        return self.value >= 40

class ScoreBook:
    def __init__(self):
        self.items = []
    def __len__(self):                 # len()
        return len(self.items)
    def __getitem__(self, idx):        # 인덱싱, for 순회
        return self.items[idx]
    def __contains__(self, subject):   # in 연산자
        return any(s.subject == subject for s in self.items)
    def __call__(self, score):         # 객체를 함수처럼 호출
        self.items.append(score)
        return self

a = Score("국어", 90)
b = Score("영어", 35)
print(a)                 # 국어 90점
print([a, b])            # [Score('국어', 90), Score('영어', 35)]
print(a + b)             # 125
print(sum([a, b]))       # 125
print(a > b, bool(b))    # True False
print(sorted([a, b]))

book = ScoreBook()
book(a)(b)               # __call__ 연쇄 호출
print(len(book), book[0], "영어" in book)
for s in book:
    print(s)
```

| 매직 메서드 | 호출 시점 |
|---|---|
| `__init__` | 객체 생성 |
| `__str__` / `__repr__` | `print()` / 대화형 출력 |
| `__len__` | `len(obj)` |
| `__getitem__` / `__setitem__` | `obj[i]` / `obj[i] = v` |
| `__add__`, `__sub__`, `__mul__` … | `+`, `-`, `*` … |
| `__eq__`, `__lt__` … | `==`, `<` … |
| `__contains__` | `in` |
| `__call__` | `obj()` |
| `__enter__` / `__exit__` | `with` 문 (6장) |
| `__iter__` / `__next__` | 반복 (8장) |

## 5.8 데이터클래스(dataclass)

```python
from dataclasses import dataclass, field

@dataclass                       # __init__, __repr__, __eq__ 자동 생성
class Item:
    name: str
    price: int
    stock: int = 0               # 기본값
    tags: list = field(default_factory=list)   # 가변 기본값은 field 사용

    def total_value(self):
        return self.price * self.stock

cola = Item("콜라", 1500, 10)
print(cola)                       # Item(name='콜라', price=1500, stock=10, tags=[])
print(cola == Item("콜라", 1500, 10))   # True
print(cola.total_value())

@dataclass(frozen=True, order=True)   # 불변 + 정렬 지원
class Point:
    x: int
    y: int
print(sorted([Point(2, 1), Point(1, 5)]))
```

## 🧪 예제 프로그램 ①: 객체지향 자판기 v3

```python
# ===== 자판기 v3: 클래스 설계 =====
from dataclasses import dataclass

@dataclass
class Beverage:
    name: str
    price: int
    stock: int

    @property
    def sold_out(self):
        return self.stock == 0

class VendingMachine:
    VALID_COINS = (50, 100, 500, 1000)

    def __init__(self):
        self._items = {}         # 번호 → Beverage
        self._balance = 0
        self._sales = 0

    def add_item(self, code, beverage):
        self._items[code] = beverage

    def insert(self, money):
        if money not in self.VALID_COINS:
            print(f"  ✗ {money}원은 사용할 수 없습니다.")
            return False
        self._balance += money
        print(f"  ＋{money}원 (잔액 {self._balance}원)")
        return True

    def select(self, code):
        item = self._items.get(code)
        if item is None:
            print("  ✗ 없는 상품 번호입니다.")
        elif item.sold_out:
            print(f"  ✗ {item.name} 품절")
        elif self._balance < item.price:
            print(f"  ✗ 잔액 부족 ({item.price - self._balance}원 더 필요)")
        else:
            item.stock -= 1
            self._balance -= item.price
            self._sales += item.price
            print(f"  ✔ {item.name} 나왔습니다. 잔액 {self._balance}원")

    def refund(self):
        change, self._balance = self._balance, 0
        print(f"  ↩ 거스름돈 {change}원")
        return change

    def __str__(self):
        lines = ["┌─── 메뉴 ───┐"]
        for code, it in self._items.items():
            state = "품절" if it.sold_out else f"{it.price}원"
            lines.append(f"[{code}] {it.name:<5} {state}")
        return "\n".join(lines)

vm = VendingMachine()
vm.add_item("1", Beverage("콜라", 1500, 2))
vm.add_item("2", Beverage("사이다", 1300, 0))
vm.add_item("3", Beverage("커피", 1000, 5))

print(vm)
for coin in (1000, 300, 500, 100):
    vm.insert(coin)
vm.select("2")
vm.select("1")
vm.select("3")
vm.refund()
```

## 5.9 모듈(Module)

**모듈**은 함수·클래스·변수를 담은 `.py` 파일 하나입니다.

**`grade_utils.py`** (모듈 파일)
```python
# grade_utils.py
"""성적 관련 유틸리티 모듈"""

PASS_SCORE = 60

def calc_grade(avg):
    for cut, g in ((90, "A"), (80, "B"), (70, "C"), (60, "D")):
        if avg >= cut:
            return g
    return "F"

def average(*scores):
    return sum(scores) / len(scores)

# 이 파일을 '직접 실행'할 때만 동작하는 테스트 코드
if __name__ == "__main__":
    print("모듈 자체 테스트:", calc_grade(average(90, 80)))
```

**`main.py`** (모듈 사용)
```python
# ① 모듈 전체 import
import grade_utils
print(grade_utils.calc_grade(85))
print(grade_utils.PASS_SCORE)

# ② 별칭
import grade_utils as gu
print(gu.average(90, 80, 70))

# ③ 특정 이름만 가져오기
from grade_utils import calc_grade, average
print(calc_grade(average(95, 90)))

# ④ 이름 바꿔 가져오기
from grade_utils import calc_grade as cg

# ⑤ 전부 가져오기 (이름 충돌 위험 → 비권장)
from grade_utils import *

# 모듈 정보
print(grade_utils.__name__)      # grade_utils
print(__name__)                  # __main__ (직접 실행된 파일)
print(dir(grade_utils))          # 모듈이 가진 이름들
```

## 5.10 패키지(Package)

**패키지**는 모듈들을 담은 **디렉터리**입니다. `__init__.py` 파일이 있으면 일반 패키지로 인식됩니다(3.3+에서는 없어도 네임스페이스 패키지로 동작).

```
calculator_app/
├── main.py
└── calc/                    ← 패키지
    ├── __init__.py
    ├── basic.py             ← 모듈
    └── advanced/            ← 하위 패키지
        ├── __init__.py
        └── science.py
```

**`calc/basic.py`**
```python
def add(a, b): return a + b
def sub(a, b): return a - b
def mul(a, b): return a * b
def div(a, b): return a / b
```

**`calc/advanced/science.py`**
```python
import math
from ..basic import mul          # 상대 import: ..은 상위 패키지

def power(a, b): return a ** b
def sqrt(a): return math.sqrt(a)
def square(a): return mul(a, a)
```

**`calc/__init__.py`** – 패키지를 import할 때 실행됨
```python
"""계산기 패키지"""
from .basic import add, sub, mul, div        # 패키지 레벨로 노출

```


## 🧪 예제 프로그램 ②: 성적표 v5 – 클래스 기반

```python
# ===== 성적표 v5 =====
from dataclasses import dataclass, field

SUBJECTS = ("국어", "영어", "수학")

@dataclass
class StudentRecord:
    name: str
    scores: dict = field(default_factory=dict)

    @property
    def total(self):
        return sum(self.scores.values())

    @property
    def average(self):
        return self.total / len(self.scores) if self.scores else 0

    @property
    def grade(self):
        for cut, g in ((90, "A"), (80, "B"), (70, "C"), (60, "D")):
            if self.average >= cut:
                return g
        return "F"

    def __lt__(self, other):              # 평균 기준 정렬
        return self.average < other.average

class ReportCard:
    def __init__(self, title):
        self.title = title
        self._students = []

    def add(self, name, *scores):
        self._students.append(StudentRecord(name, dict(zip(SUBJECTS, scores))))

    def subject_average(self, subject):
        vals = [s.scores[subject] for s in self._students]
        return sum(vals) / len(vals)

    def __len__(self):
        return len(self._students)

    def __iter__(self):                   # 평균 내림차순으로 순회
        return iter(sorted(self._students, reverse=True))

    def print(self):
        print(f"📋 {self.title} ({len(self)}명)")
        print(f"{'순위':<3}{'이름':<5}" + "".join(f"{s:>5}" for s in SUBJECTS) + f"{'평균':>7}{'학점':>4}")
        for rank, st in enumerate(self, 1):
            row = "".join(f"{st.scores[s]:>6}" for s in SUBJECTS)
            print(f"{rank:<4}{st.name:<5}{row}{st.average:>8.1f}{st.grade:>5}")
        print("과목평균  " + "".join(f"{self.subject_average(s):>6.1f}" for s in SUBJECTS))

rc = ReportCard("1학기 중간고사")
rc.add("홍길동", 90, 85, 77)
rc.add("김영희", 95, 92, 88)
rc.add("이철수", 55, 38, 70)
rc.print()
```
