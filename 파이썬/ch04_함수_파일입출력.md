## 4.1 함수 정의와 호출

```python
# def 함수이름(매개변수):  → 정의
#     """docstring: 함수 설명"""
#     실행문
#     return 반환값

def greet():
    """인사말을 출력하는 함수"""
    print("안녕하세요!")

greet()                      # 호출

def add(a, b):
    """두 수를 더해 반환"""
    return a + b

result = add(3, 5)           # 3, 5는 '인자(argument)', a, b는 '매개변수(parameter)'
print(result)                # 8

# return이 없으면 None 반환
def no_return():
    pass
print(no_return())           # None

# 여러 값 반환 → 실제로는 튜플 하나를 반환
def min_max(nums):
    return min(nums), max(nums)

lo, hi = min_max([3, 9, 1])
print(lo, hi)                # 1 9

# return은 함수를 즉시 종료 (조기 반환)
def grade(score):
    if score < 0 or score > 100:
        return "잘못된 점수"
    if score >= 90:
        return "A"
    return "B 이하"

print(grade(95), grade(150))

# docstring 확인
print(add.__doc__)
help(add)
```

## 4.2 매개변수의 종류

```python
# ① 위치 인자(positional) vs 키워드 인자(keyword)
def info(name, age, city):
    print(f"{name}({age}) - {city}")

info("홍길동", 20, "광주")                   # 위치로 전달
info(city="대전", name="김영희", age=22)     # 이름으로 전달 (순서 무관)
info("이철수", city="서울", age=25)          # 혼합 (위치 인자가 먼저)

# ② 기본값 매개변수 (기본값 있는 매개변수는 뒤쪽에)
def make_coffee(menu, size="Regular", shot=1):
    print(f"{menu} / {size} / 샷 {shot}")

make_coffee("아메리카노")
make_coffee("라떼", "Large")
make_coffee("모카", shot=2)

# ⚠️ 가변 객체를 기본값으로 쓰면 안 되는 이유
def bad_append(item, box=[]):     # 기본값 리스트는 '정의 시 한 번만' 생성됨
    box.append(item)
    return box
print(bad_append(1))   # [1]
print(bad_append(2))   # [1, 2]  ← 이전 값이 남아 있음!

def good_append(item, box=None):  # 올바른 방법
    if box is None:
        box = []
    box.append(item)
    return box
print(good_append(1), good_append(2))   # [1] [2]

# ③ 가변 위치 인자 *args → 튜플로 받음
def total(*args):
    print(type(args), args)
    return sum(args)
print(total(1, 2, 3, 4))           # 10

# ④ 가변 키워드 인자 **kwargs → 딕셔너리로 받음
def profile(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")
profile(name="홍길동", major="컴공", year=3)

# ⑤ 모두 함께 쓸 때의 순서
def func(a, b=2, *args, c, d=4, **kwargs):
    print(a, b, args, c, d, kwargs)
func(1, 3, 5, 6, c=7, e=8)   # 1 3 (5, 6) 7 4 {'e': 8}

# ⑥ 키워드 전용(*) / 위치 전용(/) 매개변수 (3.8+)
def divide(a, b, /, *, round_to=2):
    # a, b는 위치로만 / round_to는 키워드로만 전달 가능
    return round(a / b, round_to)
print(divide(10, 3, round_to=3))  # 3.333
# divide(a=10, b=3)  → TypeError

# ⑦ 인자 언패킹 * / **
nums = [1, 2, 3]
print(total(*nums))               # total(1, 2, 3)
opts = {"size": "Large", "shot": 3}
make_coffee("라떼", **opts)
```

## 4.3 변수의 유효 범위(Scope) – LEGB

파이썬은 변수를 **L**ocal → **E**nclosing → **G**lobal → **B**uilt-in 순서로 찾습니다.

```python
x = "전역(Global)"

def outer():
    x = "감싸는(Enclosing)"
    def inner():
        x = "지역(Local)"
        print(x)              # 지역(Local)
    inner()
    print(x)                  # 감싸는(Enclosing)

outer()
print(x)                      # 전역(Global)
print(len)                    # Built-in 함수

# global: 함수 안에서 전역 변수를 수정
counter = 0
def increase():
    global counter
    counter += 1
increase(); increase()
print(counter)                # 2

# nonlocal: 바깥 함수의 변수를 수정 (클로저, 8장에서 심화)
def make_counter():
    count = 0
    def inc():
        nonlocal count
        count += 1
        return count
    return inc
c = make_counter()
print(c(), c(), c())          # 1 2 3

# 💡 global 남용은 디버깅을 어렵게 함 → 매개변수/반환값 사용 권장
```

## 4.4 람다(lambda) 함수

```python
# lambda 매개변수: 표현식   (이름 없는 한 줄 함수)
square = lambda x: x ** 2
print(square(5))                     # 25

add = lambda a, b=10: a + b
print(add(1), add(1, 2))             # 11 3

# 주 용도: 다른 함수의 인자로 전달
students = [("홍길동", 90), ("김영희", 95), ("이철수", 80)]
students.sort(key=lambda s: s[1], reverse=True)   # 점수 기준 정렬
print(students)

print(list(map(lambda x: x * 2, [1, 2, 3])))      # [2, 4, 6]
print(list(filter(lambda x: x > 1, [1, 2, 3])))   # [2, 3]

# 조건 표현식도 가능
pass_fail = lambda s: "P" if s >= 60 else "F"
print(pass_fail(70))
```

## 4.5 재귀 함수(Recursion)

```python
# 자기 자신을 호출하는 함수 → 반드시 '종료 조건'이 필요
def factorial(n):
    if n <= 1:              # 종료 조건 (base case)
        return 1
    return n * factorial(n - 1)

print(factorial(5))         # 120

def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)
print([fibonacci(i) for i in range(10)])

# 리스트 합계 재귀
def rsum(lst):
    return 0 if not lst else lst[0] + rsum(lst[1:])
print(rsum([1, 2, 3, 4]))

# ⚠️ 기본 재귀 한도 약 1000 → 깊은 재귀는 반복문으로 전환하거나 메모이제이션 사용
import sys
print(sys.getrecursionlimit())
```

## 4.6 일급 객체로서의 함수

```python
# 함수는 값처럼 변수에 담고, 인자로 넘기고, 반환할 수 있다
def plus(a, b): return a + b
def minus(a, b): return a - b

op = plus                      # 변수에 대입
print(op(3, 2))                # 5

operations = {"+": plus, "-": minus}   # 딕셔너리에 저장
print(operations["-"](10, 4))  # 6

def apply(func, a, b):         # 인자로 전달 (고차 함수)
    return func(a, b)
print(apply(minus, 9, 3))

def get_op(symbol):            # 함수를 반환
    return operations[symbol]
print(get_op("+")(1, 1))

# 함수 어노테이션 (8장에서 심화)
def area(w: float, h: float) -> float:
    return w * h
print(area.__annotations__)
```

## 🧪 예제 프로그램 ①: 계산기 v3 – 함수 디스패치

```python
# ===== 계산기 v3: 함수 + 딕셔너리 디스패치 =====
def add(a, b): return a + b
def sub(a, b): return a - b
def mul(a, b): return a * b
def div(a, b):
    if b == 0:
        return None            # 예외처리는 6장에서 개선
    return a / b

OPERATORS = {
    "+": add, "-": sub, "*": mul, "/": div,
    "**": lambda a, b: a ** b,
    "%": lambda a, b: a % b,
}

def calculate(expr: str):
    """'3 + 4' 형태의 문자열을 계산"""
    parts = expr.split()
    if len(parts) != 3:
        return "형식: 숫자 연산자 숫자"
    a, op, b = parts
    func = OPERATORS.get(op)
    if func is None:
        return f"'{op}'는 지원하지 않습니다"
    result = func(float(a), float(b))
    return "0으로 나눌 수 없음" if result is None else result

def calculate_all(*exprs, show_index=True):
    """여러 식을 한 번에 계산 (가변 인자 + 키워드 전용 인자)"""
    for i, e in enumerate(exprs, 1):
        prefix = f"[{i}] " if show_index else ""
        print(f"{prefix}{e} = {calculate(e)}")

calculate_all("3 + 4", "10 / 0", "2 ** 10", "7 & 3")
```

## 4.7 파일 열기 모드

| 모드 | 의미 |
|---|---|
| `"r"` | 읽기 (기본값, 파일 없으면 오류) |
| `"w"` | 쓰기 (파일 있으면 **내용 삭제** 후 새로 씀) |
| `"a"` | 추가 (끝에 이어 씀) |
| `"x"` | 새 파일 생성 (이미 있으면 오류) |
| `"b"` | 바이너리 (`"rb"`, `"wb"`) |
| `"t"` | 텍스트 (기본값) |
| `"+"` | 읽기+쓰기 (`"r+"`) |

## 4.8 기본 파일 쓰기/읽기

```python
# ① open / close 기본형
f = open("memo.txt", "w", encoding="utf-8")   # 한글은 encoding 지정 필수
f.write("첫 번째 줄\n")
f.write("두 번째 줄\n")
f.close()                                      # 반드시 닫기

# ② with 문 (권장): 블록이 끝나면 자동으로 close
with open("memo.txt", "a", encoding="utf-8") as f:
    f.write("세 번째 줄(추가)\n")
    f.writelines(["네 번째\n", "다섯 번째\n"])  # 리스트를 한 번에

# print로 파일에 쓰기
with open("memo.txt", "a", encoding="utf-8") as f:
    print("print로 쓴 줄", file=f)

# ③ 읽기 방법 4가지
with open("memo.txt", "r", encoding="utf-8") as f:
    content = f.read()            # 전체를 하나의 문자열로
print(content)

with open("memo.txt", encoding="utf-8") as f:
    first = f.readline()          # 한 줄씩
    print("첫 줄:", first.strip())

with open("memo.txt", encoding="utf-8") as f:
    lines = f.readlines()         # 모든 줄을 리스트로
print(lines)

with open("memo.txt", encoding="utf-8") as f:
    for line in f:                # 파일 객체를 직접 순회 (메모리 효율 최고)
        print(line.rstrip("\n"))

# ④ 파일 포인터 위치
with open("memo.txt", encoding="utf-8") as f:
    f.read(3)
    print(f.tell())               # 현재 위치(바이트)
    f.seek(0)                     # 처음으로 이동
    print(f.readline().strip())

# ⑤ 여러 파일 동시에 열기
with open("memo.txt", encoding="utf-8") as src, \
     open("memo_copy.txt", "w", encoding="utf-8") as dst:
    dst.write(src.read())

# ⑥ 바이너리 파일 (이미지 복사 등)
with open("memo.txt", "rb") as f:
    data = f.read()
print(type(data), data[:10])      # bytes
```

## 4.9 CSV 파일

```python
import csv

rows = [["이름", "국어", "영어", "수학"],
        ["홍길동", 90, 85, 77],
        ["김영희", 95, 92, 88]]

# 쓰기 (newline=""은 빈 줄 방지)
with open("scores.csv", "w", newline="", encoding="utf-8-sig") as f:
    writer = csv.writer(f)
    writer.writerows(rows)        # 여러 행

# 읽기 (리스트)
with open("scores.csv", encoding="utf-8-sig") as f:
    reader = csv.reader(f)
    header = next(reader)         # 첫 줄 = 헤더
    for row in reader:
        print(row)                # 모든 값은 문자열!

# 딕셔너리로 읽기/쓰기
with open("scores.csv", encoding="utf-8-sig") as f:
    for row in csv.DictReader(f):
        print(row["이름"], int(row["국어"]))

with open("scores2.csv", "w", newline="", encoding="utf-8-sig") as f:
    w = csv.DictWriter(f, fieldnames=["이름", "평균"])
    w.writeheader()
    w.writerow({"이름": "홍길동", "평균": 84.0})
```

## 4.10 JSON 파일

```python
import json

data = {"name": "홍길동", "scores": [90, 85, 77], "pass": True, "memo": None}

s = json.dumps(data, ensure_ascii=False)
```

## 🧪 예제 프로그램 ②: 성적표 v4 – 함수 + 파일 저장/불러오기

```python
# ===== 성적표 v4 =====
import csv
import json

SUBJECTS = ("국어", "영어", "수학")

def calc_grade(avg: float) -> str:
    """평균으로 학점 계산"""
    for cut, g in ((90, "A"), (80, "B"), (70, "C"), (60, "D")):
        if avg >= cut:
            return g
    return "F"

def make_record(name, *scores):
    """학생 1명의 레코드(딕셔너리) 생성"""
    total = sum(scores)
    avg = total / len(scores)
    return {"name": name, **dict(zip(SUBJECTS, scores)),
            "total": total, "avg": round(avg, 2), "grade": calc_grade(avg)}

def save_csv(records, path):
    with open(path, "w", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=records[0].keys())
        w.writeheader()
        w.writerows(records)

def load_csv(path):
    with open(path, encoding="utf-8-sig") as f:
        return list(csv.DictReader(f))

def save_json(records, path):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(records, f, ensure_ascii=False, indent=2)

def print_report(records):
    print(f"{'이름':<5}{'총점':>6}{'평균':>8}{'학점':>4}")
    for r in sorted(records, key=lambda r: float(r["avg"]), reverse=True):
        print(f"{r['name']:<5}{r['total']:>7}{float(r['avg']):>9.2f}{r['grade']:>5}")

# ----- 실행 -----
records = [make_record("홍길동", 90, 85, 77),
           make_record("김영희", 95, 92, 88),
           make_record("이철수", 55, 38, 70)]

save_csv(records, "report.csv")
save_json(records, "report.json")

loaded = load_csv("report.csv")       # CSV에서 다시 읽기
print_report(loaded)
```
