## 📁 backend/main.py

```python
from fastapi import FastAPI, UploadFile, File
from fastapi.responses import StreamingResponse, JSONResponse
from ultralytics import YOLO
import cv2, time
import numpy as np
from pymongo import MongoClient
from datetime import datetime

app = FastAPI()
model = YOLO("yolov8n.pt")

client = MongoClient("mongodb://localhost:27017")
db = client["vision_db"]
collection = db["detections"]

latest_frame = None
latest_results = []
alarm_config = {"class": "person", "threshold": 1}

@app.post("/alarm/config")
def set_alarm(config: dict):
    global alarm_config
    alarm_config = config
    return {"status": "alarm updated"}

@app.post("/upload")
async def upload(file: UploadFile = File(...)):
    global latest_frame, latest_results

    start = time.time()
    data = await file.read()
    frame = cv2.imdecode(np.frombuffer(data, np.uint8), cv2.IMREAD_COLOR)

    results = model(frame)[0]
    detections = []

    for box in results.boxes:
        cls = model.names[int(box.cls)]
        conf = float(box.conf)
        x1, y1, x2, y2 = map(int, box.xyxy[0])

        detections.append({
            "class": cls,
            "confidence": round(conf,2),
            "box": [x1,y1,x2,y2],
            "time": datetime.now()
        })

    fps = round(1/(time.time()-start),2)

    # MongoDB 저장
    collection.insert_many(detections) if detections else None

    # 알람 조건 확인
    alarm = False
    count = sum(1 for d in detections if d["class"] == alarm_config["class"])
    if count >= alarm_config["threshold"]:
        alarm = True

    latest_results = detections
    annotated = results.plot()
    latest_frame = annotated

    return JSONResponse({"fps": fps, "detections": detections, "alarm": alarm})


def gen_frames():
    while True:
        if latest_frame is not None:
            _, buffer = cv2.imencode(".jpg", latest_frame)
            yield (b"--frame\r\nContent-Type:image/jpeg\r\n\r\n" + buffer.tobytes() + b"\r\n")

@app.get("/video")
def video():
    return StreamingResponse(gen_frames(), media_type="multipart/x-mixed-replace; boundary=frame")


@app.get("/stats")
def stats():
    pipeline = [
        {"$group": {
            "_id": {"hour": {"$hour": "$time"}, "class": "$class"},
            "count": {"$sum":1}
        }},
        {"$sort": {"_id.hour":1}}
    ]
    data = list(collection.aggregate(pipeline))
    return data
```

# ✅ 2️⃣ Sender (영상 송신)

```python
import cv2, requests

cap = cv2.VideoCapture(0)
url = "http://localhost:8000/upload"

while True:
    ret, frame = cap.read()
    _, img = cv2.imencode(".jpg", frame)
    files = {"file": img.tobytes()}
    requests.post(url, files=files)

    cv2.imshow("Sender", frame)
    if cv2.waitKey(1)==27:
        break
```

## 📁 src/App.svelte

```svelte
<script>
  import { onMount } from "svelte";
  import Chart from "chart.js/auto";

  let videoUrl = "http://localhost:8000/video";
  let results = [];
  let fps = 0;
  let alarm = false;

  let alarmClass = "person";
  let threshold = 1;

  let chart;

  async function fetchStats() {
    const res = await fetch("http://localhost:8000/stats");
    const data = await res.json();

    const labels = data.map(d => d._id.hour + "h");
    const values = data.map(d => d.count);

    chart.data.labels = labels;
    chart.data.datasets[0].data = values;
    chart.update();
  }

  async function setAlarm() {
    await fetch("http://localhost:8000/alarm/config", {
      method: "POST",
      headers: {"Content-Type":"application/json"},
      body: JSON.stringify({class: alarmClass, threshold})
    });
  }

  onMount(() => {
    const ctx = document.getElementById("chart");
    chart = new Chart(ctx, {
      type: "bar",
      data: { labels: [], datasets:[{label:"Objects", data: []}] }
    });

    setInterval(fetchStats, 5000);
  });
</script>

<h2>🎥 실시간 객체 검출</h2>
<img src={videoUrl} width="640">

<h3>FPS: {fps}</h3>

{#if alarm}
<h2 style="color:red;">🚨 ALARM DETECTED!</h2>
{/if}

<h3>🔔 Alarm 설정</h3>
<select bind:value={alarmClass}>
  <option>person</option>
  <option>car</option>
  <option>dog</option>
</select>
<input type="number" bind:value={threshold}>
<button on:click={setAlarm}>적용</button>

<h3>📋 Detection Table</h3>
<table border="1">
<tr><th>Class</th><th>Confidence</th></tr>
{#each results as r}
<tr>
<td>{r.class}</td>
<td>{r.confidence}</td>
</tr>
{/each}
</table>

<h3>📈 누적 통계</h3>
<canvas id="chart"></canvas>
```
