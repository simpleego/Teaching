# ✅ 2️⃣ Svelte 프론트엔드 전체 코드 (차트 + 알람 추가)

📄 `frontend/src/App.svelte`

```svelte
<script>
  import { onMount } from "svelte";
  import Chart from "chart.js/auto";

  let socket;
  let imageSrc = "";
  let detections = [];

  let showBoxes = true;
  let fps = 0;
  let lastTime = performance.now();

  let alertObject = "person";   // 알람 대상 객체
  let alertMessage = "";

  let chart;
  let chartData = {};

  let canvas;

  function updateChart(detections) {
    chartData = {};

    detections.forEach(d => {
      chartData[d.label] = (chartData[d.label] || 0) + 1;
    });

    const labels = Object.keys(chartData);
    const values = Object.values(chartData);

    chart.data.labels = labels;
    chart.data.datasets[0].data = values;
    chart.update();
  }

  function checkAlert(detections) {
    const found = detections.find(d => d.label === alertObject);
    if (found) {
      alertMessage = `🚨 ${alertObject} detected!`;
      playSound();
    } else {
      alertMessage = "";
    }
  }

  function playSound() {
    const audio = new Audio("https://actions.google.com/sounds/v1/alarms/beep_short.ogg");
    audio.play();
  }

  onMount(() => {
    chart = new Chart(canvas, {
      type: "bar",
      data: {
        labels: [],
        datasets: [{
          label: "Object Count",
          data: [],
        }]
      }
    });

    socket = new WebSocket("ws://localhost:8000/ws");

    socket.onmessage = (event) => {
      const now = performance.now();
      fps = Math.round(1000 / (now - lastTime));
      lastTime = now;

      const data = JSON.parse(event.data);
      imageSrc = "data:image/jpeg;base64," + data.image;
      detections = data.detections;

      updateChart(detections);
      checkAlert(detections);
    };
  });
</script>

<main>
  <h2>📡 Object Detection Dashboard</h2>

  <!-- Control Panel -->
  <div class="controls">
    <button on:click={() => showBoxes = !showBoxes}>
      {showBoxes ? "박스 OFF" : "박스 ON"}
    </button>
    <span>FPS: {fps}</span>
  </div>

  <!-- Alert -->
  {#if alertMessage}
    <div class="alert">{alertMessage}</div>
  {/if}

  <!-- Video -->
  <div class="video-container">
    <img src={imageSrc} width="640" height="480" />
    {#if showBoxes}
      {#each detections as d}
        <div class="box"
          style="
            left:{d.x1}px;
            top:{d.y1}px;
            width:{d.x2 - d.x1}px;
            height:{d.y2 - d.y1}px;
          ">
          <span>{d.label} ({d.confidence})</span>
        </div>
      {/each}
    {/if}
  </div>

  <!-- Detection Table -->
  <h3>📋 Detection Results</h3>
  <table>
    <thead>
      <tr>
        <th>No</th><th>Label</th><th>Confidence</th>
        <th>x1</th><th>y1</th><th>x2</th><th>y2</th>
      </tr>
    </thead>
    <tbody>
      {#each detections as d, i}
        <tr>
          <td>{i+1}</td>
          <td>{d.label}</td>
          <td>{d.confidence}</td>
          <td>{d.x1}</td>
          <td>{d.y1}</td>
          <td>{d.x2}</td>
          <td>{d.y2}</td>
        </tr>
      {/each}
    </tbody>
  </table>

  <!-- Chart -->
  <h3>📊 Object Statistics</h3>
  <canvas bind:this={canvas} width="400" height="200"></canvas>
</main>

<style>
main { text-align: center; }

.controls { margin-bottom: 10px; }

.video-container {
  position: relative;
  width: 640px;
  height: 480px;
  margin: auto;
}

img { position: absolute; }

.box {
  position: absolute;
  border: 2px solid red;
}
.box span {
  background: rgba(0,0,0,0.6);
  color: white;
  font-size: 12px;
}

.alert {
  color: white;
  background: red;
  padding: 10px;
  margin: 10px;
  font-weight: bold;
}

table {
  margin: 20px auto;
  border-collapse: collapse;
  width: 80%;
}
th, td {
  border: 1px solid #aaa;
  padding: 5px;
}
th { background: #eee; }
</style>
```

# ✅ 추가 기능 설명

## 📊 1. 객체 통계 차트

* detections 배열을 집계:

```js
chartData[d.label] = (chartData[d.label] || 0) + 1;
```

* Chart.js Bar Chart로 표시

예:

```
person: 2
car: 1
dog: 1
```

---

## 🚨 2. 특정 객체 감지 알람

```js
if (d.label === "person") {
   alertMessage = "🚨 person detected!";
   playSound();
}
```

* 화면 경고 메시지 표시
* 소리 알림(beep)
* 원하는 객체명으로 변경 가능:

```js
let alertObject = "car";
```
