# 📁 backend/main.py

```python
from fastapi import FastAPI, WebSocket
from database import collection
from datetime import datetime
import random, asyncio

app = FastAPI()

objects = ["person", "car", "dog"]

thresholds = {"person": 3, "car": 2, "dog": 1}

@app.websocket("/ws/detect")
async def ws_detect(ws: WebSocket):
    await ws.accept()
    while True:
        obj = random.choice(objects)
        data = {
            "time": datetime.now(),
            "object": obj,
            "confidence": round(random.uniform(0.5, 0.99),2)
        }

        await collection.insert_one(data)

        count = await collection.count_documents({"object": obj})

        alarm = count >= thresholds[obj]

        await ws.send_json({
            "time": data["time"].strftime("%H:%M:%S"),
            "object": obj,
            "confidence": data["confidence"],
            "alarm": alarm
        })

        await asyncio.sleep(1)


@app.get("/stats")
async def stats():
    pipeline = [
        {"$group": {"_id": "$object", "count": {"$sum":1}}},
    ]
    result=[]
    async for doc in collection.aggregate(pipeline):
        result.append({"object":doc["_id"],"count":doc["count"]})
    return result


@app.get("/timeline")
async def timeline():
    pipeline = [
        {"$group": {
            "_id": {"hour":{"$hour":"$time"}},
            "count":{"$sum":1}
        }},
        {"$sort":{"_id.hour":1}}
    ]

    result=[]
    async for doc in collection.aggregate(pipeline):
        result.append({"time":doc["_id"]["hour"],"count":doc["count"]})
    return result


@app.post("/threshold/{obj}/{value}")
async def set_threshold(obj:str,value:int):
    thresholds[obj]=value
    return thresholds
```

# 📁 frontend/src/App.svelte

```svelte
<script>
  import { onMount } from "svelte";
  import Chart from "chart.js/auto";

  let socket;
  let table = [];
  let labels = [];
  let counts = [];

  let alarmMsg = "";

  let chart;

  let selectedObj="person";
  let threshold=3;

  onMount(() => {
    const ctx=document.getElementById("chart");
    chart = new Chart(ctx,{
      type:"bar",
      data:{labels:[],datasets:[{label:"Object Count",data:[]}]},
    });

    socket=new WebSocket("ws://localhost:8000/ws/detect");
    socket.onmessage=(e)=>{
      const data=JSON.parse(e.data);

      table.unshift(data);
      if(table.length>10) table.pop();

      if(data.alarm){
        alarmMsg=`🚨 ${data.object} threshold exceeded!`;
      }

      loadStats();
    };
  });

  async function loadStats(){
    const res=await fetch("http://localhost:8000/stats");
    const stats=await res.json();
    labels=stats.map(s=>s.object);
    counts=stats.map(s=>s.count);

    chart.data.labels=labels;
    chart.data.datasets[0].data=counts;
    chart.update();
  }

  async function setThreshold(){
    await fetch(`http://localhost:8000/threshold/${selectedObj}/${threshold}`,{method:"POST"});
    alert("Threshold Updated");
  }
</script>

<h2>📊 객체 감지 통합 대시보드</h2>

<div>
<select bind:value={selectedObj}>
<option>person</option>
<option>car</option>
<option>dog</option>
</select>
<input type="number" bind:value={threshold}/>
<button on:click={setThreshold}>알람 설정</button>
</div>

<h3 style="color:red">{alarmMsg}</h3>

<canvas id="chart" width="500" height="250"></canvas>

<h3>실시간 JSON 테이블</h3>
<table border="1">
<tr><th>Time</th><th>Object</th><th>Conf</th><th>Alarm</th></tr>
{#each table as row}
<tr>
<td>{row.time}</td>
<td>{row.object}</td>
<td>{row.confidence}</td>
<td>{row.alarm}</td>
</tr>
{/each}
</table>
```
