아래는 요청하신
✅ **10단계 Chart.js 학습용 통합 프로젝트 (FastAPI + Svelte)**
를 하나의 완성 프로젝트 형태로 제공합니다.

* FastAPI에서 **step1 ~ step10 API 제공**

# 📦 Backend (FastAPI)

## backend/main.py

```python
from fastapi import FastAPI
import random
from datetime import datetime

app = FastAPI()

@app.get("/step1")
def step1():
    return [10,20,30,40]

@app.get("/step2")
def step2():
    return [{"name":"Apple","value":30},{"name":"Banana","value":20}]

@app.get("/step3")
def step3():
    return {"2024":[10,20,30],"2025":[15,25,35]}

@app.get("/step4")
def step4():
    return [
        {"time":"10:00","value":5},
        {"time":"11:00","value":9},
        {"time":"12:00","value":15}
    ]

@app.get("/step5")
def step5():
    return {"person":12,"car":8,"dog":3}

@app.get("/step6")
def step6():
    return [5,10,20,35,50]

@app.get("/step7")
def step7():
    return random.randint(0,100)

@app.get("/step8")
def step8():
    return {"speed":70,"power":90,"accuracy":85,"stability":60}

@app.get("/step9")
def step9(type:str="day"):
    if type=="day":
        return [5,10,20]
    return [20,30,40]

@app.get("/step10")
def step10():
    return [
        {"class":"person","count":10},
        {"class":"car","count":5},
        {"class":"dog","count":3}
    ]
```

실행:

```bash
uvicorn main:app --reload
```

## frontend/src/App.svelte

```svelte
<script>
  import { onMount } from "svelte";
  import Chart from "chart.js/auto";

  async function loadChart(id, url, type, mapFn) {
    const res = await fetch(url);
    const data = await res.json();
    const {labels, values, datasets} = mapFn(data);

    new Chart(document.getElementById(id), {
      type,
      data: {
        labels,
        datasets: datasets ?? [{label:id, data: values}]
      }
    });
  }

  onMount(async () => {
    // STEP1
    loadChart("c1","http://localhost:8000/step1","bar",
      d=>({labels:["A","B","C","D"], values:d}));

    // STEP2
    loadChart("c2","http://localhost:8000/step2","pie",
      d=>({labels:d.map(x=>x.name), values:d.map(x=>x.value)}));

    // STEP3
    loadChart("c3","http://localhost:8000/step3","bar",
      d=>({
        labels:["Q1","Q2","Q3"],
        datasets:[
          {label:"2024",data:d["2024"]},
          {label:"2025",data:d["2025"]}
        ]
      }));

    // STEP4
    loadChart("c4","http://localhost:8000/step4","line",
      d=>({labels:d.map(x=>x.time), values:d.map(x=>x.value)}));

    // STEP5
    loadChart("c5","http://localhost:8000/step5","bar",
      d=>({labels:Object.keys(d), values:Object.values(d)}));

    // STEP6
    loadChart("c6","http://localhost:8000/step6","line",
      d=>({labels:["1","2","3","4","5"], values:d}));

    // STEP7 (real-time)
    const ctx7 = document.getElementById("c7");
    const chart7 = new Chart(ctx7,{
      type:"line",
      data:{labels:[],datasets:[{label:"Realtime",data:[]}]}
    });

    setInterval(async()=>{
      const v = await fetch("http://localhost:8000/step7").then(r=>r.json());
      chart7.data.labels.push(new Date().toLocaleTimeString());
      chart7.data.datasets[0].data.push(v);
      if(chart7.data.labels.length>10){
        chart7.data.labels.shift();
        chart7.data.datasets[0].data.shift();
      }
      chart7.update();
    },1000);

    // STEP8
    loadChart("c8","http://localhost:8000/step8","radar",
      d=>({labels:Object.keys(d), values:Object.values(d)}));

    // STEP9
    loadChart("c9","http://localhost:8000/step9?type=day","bar",
      d=>({labels:["A","B","C"], values:d}));

    // STEP10
    const tableData = await fetch("http://localhost:8000/step10").then(r=>r.json());
    document.getElementById("table").innerHTML =
      tableData.map(d=>`<tr><td>${d.class}</td><td>${d.count}</td></tr>`).join("");

    loadChart("c10","http://localhost:8000/step10","bar",
      d=>({labels:d.map(x=>x.class), values:d.map(x=>x.count)}));
  });
</script>
```
