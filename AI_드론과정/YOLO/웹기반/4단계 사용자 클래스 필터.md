가능합니다. **“클래스 필터를 사용자(UI)에서 선택 → 서버에 전달 → 서버가 해당 클래스만 검출 결과로 push”** 형태로 바꾸면 됩니다.
구현은 WebSocket이 가장 자연스럽습니다(클라이언트가 연결 중에도 필터를 바꿔야 하니까요).

아래는 **web_server_stream.py 개선(클래스 필터 지원)** + **Svelte 프론트 수정(멀티 셀렉트 리스트박스 + WS로 필터 전송)** 입니다.

# 1) web_server_stream.py 개선본 (클래스 필터 지원)

핵심 변경점

* `/classes` 엔드포인트 추가: 프론트가 모델 클래스 목록을 받아 UI를 구성
* WebSocket 연결마다 **필터 상태(ws별 선택 class_id set)** 저장
* 클라이언트가 WS로 `{type:"set_filter", class_ids:[...]}` 전송하면 즉시 반영
* inference loop는 1번만 추론하고, **각 클라이언트에게 필터된 결과만 개별 전송**

> 아래 코드는 기존 `web_server_stream.py`를 대체하는 형태로 제공 (그대로 저장해서 실행)

```python
import asyncio
import time
import threading
from typing import Dict, Any, List, Optional, Set

import cv2
import numpy as np
import requests
from fastapi import FastAPI, WebSocket, WebSocketDisconnect, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import Response, StreamingResponse, JSONResponse

from ultralytics import YOLO


# =========================
# 설정
# =========================
CAM_SERVER_MJPEG = "http://localhost:9001/mjpeg"

MODEL_PATH = "yolov8n.pt"
CONF_THRES = 0.25
IOU_THRES = 0.45

INFER_FPS = 5.0
MAX_FRAME_AGE_SEC = 2.0
RECONNECT_DELAY_SEC = 1.0
MJPEG_READ_TIMEOUT = 10

app = FastAPI()
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173", "http://127.0.0.1:5173"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

model = YOLO(MODEL_PATH)


# =========================
# 최신 프레임 캐시
# =========================
_lock = threading.Lock()
_latest_bgr: Optional[np.ndarray] = None
_latest_jpeg: Optional[bytes] = None
_latest_ts: float = 0.0


# =========================
# WebSocket 클라이언트 + 필터 상태
# =========================
_ws_lock = asyncio.Lock()
_ws_clients: Set[WebSocket] = set()

# ws별 필터: None이면 "전체 클래스", set이면 선택된 class_id만
_ws_filters: Dict[WebSocket, Optional[Set[int]]] = {}


# =========================
# MJPEG -> JPEG 프레임 추출
# =========================
def iter_jpeg_frames_from_mjpeg(url: str):
    with requests.get(url, stream=True, timeout=MJPEG_READ_TIMEOUT) as r:
        r.raise_for_status()
        buf = bytearray()
        for chunk in r.iter_content(chunk_size=4096):
            if not chunk:
                continue
            buf.extend(chunk)

            while True:
                soi = buf.find(b"\xff\xd8")
                if soi == -1:
                    if len(buf) > 2_000_000:
                        buf.clear()
                    break
                eoi = buf.find(b"\xff\xd9", soi + 2)
                if eoi == -1:
                    break
                jpg = bytes(buf[soi:eoi + 2])
                del buf[:eoi + 2]
                yield jpg


def frame_cache_worker():
    global _latest_bgr, _latest_jpeg, _latest_ts
    while True:
        try:
            for jpg in iter_jpeg_frames_from_mjpeg(CAM_SERVER_MJPEG):
                arr = np.frombuffer(jpg, dtype=np.uint8)
                bgr = cv2.imdecode(arr, cv2.IMREAD_COLOR)
                if bgr is None:
                    continue
                with _lock:
                    _latest_bgr = bgr
                    _latest_jpeg = jpg
                    _latest_ts = time.time()
        except Exception as e:
            print(f"[ingest] stream error: {e} -> reconnecting in {RECONNECT_DELAY_SEC}s")
            time.sleep(RECONNECT_DELAY_SEC)


def get_latest_frame(max_age_sec: float = MAX_FRAME_AGE_SEC) -> np.ndarray:
    with _lock:
        bgr = None if _latest_bgr is None else _latest_bgr.copy()
        ts = _latest_ts
    if bgr is None:
        raise HTTPException(status_code=503, detail="No frame available yet.")
    if (time.time() - ts) > max_age_sec:
        raise HTTPException(status_code=503, detail="Frame too old. Stream may be down.")
    return bgr


@app.on_event("startup")
def startup():
    threading.Thread(target=frame_cache_worker, daemon=True).start()
    loop = asyncio.get_event_loop()
    loop.create_task(inference_loop())


# =========================
# 프론트 UI용: 모델 클래스 목록 제공
# =========================
@app.get("/classes")
def classes():
    # Ultralytics model.names: {id: name} 형태가 많음
    names = model.names
    items = [{"id": int(k), "name": str(v)} for k, v in names.items()]
    items.sort(key=lambda x: x["id"])
    return JSONResponse({"model": MODEL_PATH, "classes": items})


# =========================
# (선택) 영상 미리보기: MJPEG 재포장
# =========================
@app.get("/video")
def video():
    def gen():
        boundary = b"frame"
        while True:
            with _lock:
                jpg = _latest_jpeg
            if jpg is None:
                time.sleep(0.05)
                continue
            yield b"--" + boundary + b"\r\n"
            yield b"Content-Type: image/jpeg\r\n\r\n"
            yield jpg
            yield b"\r\n"
            time.sleep(1 / 15)

    return StreamingResponse(gen(), media_type="multipart/x-mixed-replace; boundary=frame")


@app.get("/frame.jpg")
def frame_jpg():
    with _lock:
        jpg = _latest_jpeg
        ts = _latest_ts
    if jpg is None:
        raise HTTPException(status_code=503, detail="No frame available yet.")
    if (time.time() - ts) > MAX_FRAME_AGE_SEC:
        raise HTTPException(status_code=503, detail="Frame too old.")
    return Response(content=jpg, media_type="image/jpeg")


# =========================
# WebSocket: /detect_stream
#  - 클라이언트가 set_filter를 보내면 ws별 필터 갱신
# =========================
@app.websocket("/detect_stream")
async def detect_stream(ws: WebSocket):
    await ws.accept()
    async with _ws_lock:
        _ws_clients.add(ws)
        _ws_filters[ws] = None  # 기본: 전체 클래스

    # 연결 직후 현재 필터 상태/클래스 목록 일부 안내
    await ws.send_json({
        "type": "hello",
        "ts": time.time(),
        "filter": None,
        "hint": "Send {type:'set_filter', class_ids:[0,1,...]} to filter, or []/null to reset(all)."
    })

    try:
        # 클라이언트 메시지 수신 루프 (필터 변경 반영)
        while True:
            msg = await ws.receive_json()
            if not isinstance(msg, dict):
                continue

            if msg.get("type") == "set_filter":
                raw = msg.get("class_ids", None)

                # None 또는 [] => 전체로 리셋
                if raw is None or (isinstance(raw, list) and len(raw) == 0):
                    async with _ws_lock:
                        _ws_filters[ws] = None
                    await ws.send_json({"type": "filter_ack", "ok": True, "filter": None, "ts": time.time()})
                    continue

                if not isinstance(raw, list):
                    await ws.send_json({"type": "filter_ack", "ok": False, "detail": "class_ids must be a list", "ts": time.time()})
                    continue

                # 숫자 변환/검증
                try:
                    ids = set(int(x) for x in raw)
                except Exception:
                    await ws.send_json({"type": "filter_ack", "ok": False, "detail": "class_ids must be int list", "ts": time.time()})
                    continue

                valid_ids = set(model.names.keys())
                ids = ids.intersection(valid_ids)

                async with _ws_lock:
                    _ws_filters[ws] = ids

                await ws.send_json({"type": "filter_ack", "ok": True, "filter": sorted(list(ids)), "ts": time.time()})

    except WebSocketDisconnect:
        pass
    finally:
        async with _ws_lock:
            _ws_clients.discard(ws)
            _ws_filters.pop(ws, None)


# =========================
# ws별 개별 전송 (필터 적용)
# =========================
async def send_to_ws(ws: WebSocket, payload: Dict[str, Any]):
    try:
        await ws.send_json(payload)
        return True
    except Exception:
        return False


async def inference_loop():
    interval = 1.0 / max(INFER_FPS, 0.1)

    while True:
        t0 = time.time()

        async with _ws_lock:
            clients = list(_ws_clients)

        # 클라이언트 없으면 쉼(자원 절감)
        if not clients:
            await asyncio.sleep(0.2)
            continue

        # 최신 프레임 확보
        try:
            bgr = get_latest_frame()
        except HTTPException as e:
            # 상태 메시지는 모든 클라이언트에 동일 전송
            dead = []
            for ws in clients:
                ok = await send_to_ws(ws, {"type": "status", "ok": False, "detail": e.detail, "ts": time.time()})
                if not ok:
                    dead.append(ws)
            if dead:
                async with _ws_lock:
                    for d in dead:
                        _ws_clients.discard(d)
                        _ws_filters.pop(d, None)

            await asyncio.sleep(0.5)
            continue

        # YOLO 추론: event loop 블로킹 방지
        results = await asyncio.to_thread(
            model.predict,
            bgr,
            CONF_THRES,
            IOU_THRES,
            False
        )
        r = results[0]

        # 전체 detection 한번만 만들기
        all_dets: List[Dict[str, Any]] = []
        if r.boxes is not None and len(r.boxes) > 0:
            boxes = r.boxes.xyxy.cpu().numpy()
            confs = r.boxes.conf.cpu().numpy()
            clss = r.boxes.cls.cpu().numpy().astype(int)

            for (x1, y1, x2, y2), conf, cls_id in zip(boxes, confs, clss):
                all_dets.append({
                    "class_id": int(cls_id),
                    "class_name": model.names[int(cls_id)],
                    "confidence": float(conf),
                    "bbox_xyxy": [float(x1), float(y1), float(x2), float(y2)]
                })

        # ws별 필터 적용 후 개별 전송
        dead = []
        for ws in clients:
            async with _ws_lock:
                flt = _ws_filters.get(ws, None)  # None이면 전체

            if flt is None:
                dets = all_dets
            else:
                dets = [d for d in all_dets if d["class_id"] in flt]

            payload = {
                "type": "detections",
                "ts": time.time(),
                "image": {"width": int(bgr.shape[1]), "height": int(bgr.shape[0])},
                "filter": None if flt is None else sorted(list(flt)),
                "detections": dets
            }

            ok = await send_to_ws(ws, payload)
            if not ok:
                dead.append(ws)

        if dead:
            async with _ws_lock:
                for d in dead:
                    _ws_clients.discard(d)
                    _ws_filters.pop(d, None)

        # fps 맞추기
        dt = time.time() - t0
        if dt < interval:
            await asyncio.sleep(interval - dt)
```

# 2) Svelte 프론트 수정 (리스트박스 멀티선택 → WS로 class_ids 전송)

핵심 변경점

* `/classes` 호출해서 클래스 목록을 가져옴
* `<select multiple>` 로 멀티 선택 UI 구성
* 선택 변경 시 WS에 `{type:"set_filter", class_ids:[...]}` 전송
* 재연결 시 마지막 선택값을 다시 전송

## App.svelte (박스 Canvas 오버레이 + 클래스 필터 UI 포함)

```svelte
<script>
  import { onMount } from "svelte";

  const apiBase = "http://localhost:8000";
  const wsUrl = "ws://localhost:8000/detect_stream";

  let liveImg;
  let canvas;
  let ctx;

  let status = "disconnected";
  let lastMsg = null;

  // 클래스 목록 / 선택
  let classes = [];            // [{id,name}, ...]
  let selectedIds = [];        // [int, int, ...]  (멀티 선택)
  let ws = null;

  function resizeCanvasToImage() {
    if (!liveImg || !canvas) return;
    canvas.width = liveImg.clientWidth;
    canvas.height = liveImg.clientHeight;
  }

  function drawBoxes(msg) {
    if (!ctx || !liveImg) return;
    resizeCanvasToImage();

    const iw = msg.image.width;
    const ih = msg.image.height;
    const cw = canvas.width;
    const ch = canvas.height;
    const sx = cw / iw;
    const sy = ch / ih;

    ctx.clearRect(0, 0, cw, ch);
    ctx.lineWidth = 2;
    ctx.font = "14px sans-serif";

    for (const d of msg.detections) {
      const [x1, y1, x2, y2] = d.bbox_xyxy;
      const rx1 = x1 * sx, ry1 = y1 * sy;
      const rw = (x2 - x1) * sx, rh = (y2 - y1) * sy;

      ctx.strokeStyle = "lime";
      ctx.strokeRect(rx1, ry1, rw, rh);

      const label = `${d.class_name} ${d.confidence.toFixed(2)}`;
      ctx.fillStyle = "lime";
      const ty = Math.max(14, ry1 - 6);
      ctx.fillText(label, rx1 + 2, ty);
    }
  }

  async function loadClasses() {
    const res = await fetch(`${apiBase}/classes`);
    const json = await res.json();
    classes = json.classes || [];
  }

  function sendFilter() {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;

    // 선택이 비어있으면 "전체"로 리셋
    const payload = {
      type: "set_filter",
      class_ids: selectedIds.length ? selectedIds : null
    };
    ws.send(JSON.stringify(payload));
  }

  function connectWS() {
    ws = new WebSocket(wsUrl);
    status = "connecting";

    ws.onopen = () => {
      status = "connected";
      // 연결되면 현재 선택 필터를 즉시 적용
      sendFilter();
    };

    ws.onmessage = (ev) => {
      const msg = JSON.parse(ev.data);
      lastMsg = msg;

      if (msg.type === "detections") {
        drawBoxes(msg);
      }
    };

    ws.onclose = () => {
      status = "disconnected";
      setTimeout(connectWS, 1000);
    };
  }

  function onSelectChange(e) {
    // multiple select에서 선택값 추출
    const opts = Array.from(e.target.selectedOptions);
    selectedIds = opts.map(o => Number(o.value));
    sendFilter();
  }

  function resetFilterAll() {
    selectedIds = [];
    sendFilter();
  }

  onMount(async () => {
    ctx = canvas.getContext("2d");
    await loadClasses();
    connectWS();
    window.addEventListener("resize", resizeCanvasToImage);
    return () => window.removeEventListener("resize", resizeCanvasToImage);
  });
</script>

<main style="padding:16px; display:grid; gap:12px;">
  <h1>Real-time YOLO Stream (Class Filter)</h1>
  <div>WS Status: <b>{status}</b></div>

  <section style="display:flex; gap:12px; align-items:flex-start; flex-wrap:wrap;">
    <div style="min-width:260px;">
      <div style="display:flex; justify-content:space-between; align-items:center; gap:8px;">
        <h3 style="margin:0;">클래스 필터</h3>
        <button on:click={resetFilterAll}>전체</button>
      </div>

      <div style="font-size:12px; color:#555; margin:6px 0;">
        여러 개 선택 가능 (Ctrl/Shift)
      </div>

      <select multiple size="14" style="width:100%;"
              on:change={onSelectChange}>
        {#each classes as c}
          <option value={c.id} selected={selectedIds.includes(c.id)}>
            {c.id} - {c.name}
          </option>
        {/each}
      </select>

      <div style="font-size:12px; color:#555; margin-top:6px;">
        선택됨: {selectedIds.length ? selectedIds.join(", ") : "전체"}
      </div>
    </div>

    <div style="position:relative; width:min(960px, 100%); flex:1;">
      <img
        bind:this={liveImg}
        src={`${apiBase}/video`}
        alt="live"
        style="width:100%; display:block; border:1px solid #ccc; border-radius:8px;"
        on:load={resizeCanvasToImage}
      />
      <canvas
        bind:this={canvas}
        style="position:absolute; left:0; top:0; width:100%; height:100%; pointer-events:none;"
      ></canvas>
    </div>
  </section>

  <section>
    <h3>Last message</h3>
    <pre style="white-space:pre-wrap; background:#f7f7f7; padding:12px; border-radius:8px;">
{lastMsg ? JSON.stringify(lastMsg, null, 2) : "waiting..."}
    </pre>
  </section>
</main>
```

# 3) 동작 방식 정리

* 사용자가 리스트박스에서 클래스 선택
* 프론트가 WS로 `set_filter` 전송
* 서버는 해당 WS(사용자)에만 그 필터를 적용해서 결과를 push
  (즉, **사용자별로 서로 다른 필터를 동시에 적용 가능**)

---

# 4) 운영 팁 (성능/안정성)

* 클래스 필터는 “추론을 줄이진 않습니다.”
  (YOLO는 전체를 검출한 뒤 필터링)
  성능을 더 올리려면:

  * `INFER_FPS` 낮추기(예: 2~5)
  * 추론 입력 프레임 리사이즈(예: 640x360) 후 추론
  * 더 작은 모델 사용(yolov8n 유지)
