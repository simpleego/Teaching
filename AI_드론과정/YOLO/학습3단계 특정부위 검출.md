가능합니다. “사람의 특정 부위(머리/손/발 등)”를 **YOLO로 직접 ‘부위 클래스’로 검출**하려면 보통 2가지 접근이 있습니다.

1. **YOLO Pose(키포인트)** 로 부위 위치를 찾는다 (추천, 데이터 없이 바로 가능)
2. “hand/head/face…” 같은 **부위 라벨 데이터셋으로 YOLO를 따로 학습**한다 (정밀하지만 학습 필요)

여기서는 **1) YOLOv8 Pose 기반**으로, 사람의 키포인트를 이용해 **부위별 바운딩박스(머리/왼팔/오른팔/왼다리/오른다리/손/발 등)** 를 만들어서 화면에 그리는 파이썬 예제를 드릴게요.

# 2) 부위별 검출(키포인트 → 부위 박스 생성) 코드

* 입력: 이미지/웹캠
* 출력: 사람별 부위 박스 + 라벨 표시

```python
from ultralytics import YOLO
import cv2
import numpy as np

# -----------------------------
# COCO 17 keypoints index (YOLOv8 pose 기본)
# 0:nose, 1:left_eye, 2:right_eye, 3:left_ear, 4:right_ear,
# 5:left_shoulder, 6:right_shoulder, 7:left_elbow, 8:right_elbow,
# 9:left_wrist, 10:right_wrist, 11:left_hip, 12:right_hip,
# 13:left_knee, 14:right_knee, 15:left_ankle, 16:right_ankle
# -----------------------------

def xywh_from_points(pts, img_w, img_h, pad=10):
    """pts: (N,2) valid points in pixel coords -> xyxy box"""
    if pts.size == 0:
        return None
    x_min = max(int(np.min(pts[:, 0]) - pad), 0)
    y_min = max(int(np.min(pts[:, 1]) - pad), 0)
    x_max = min(int(np.max(pts[:, 0]) + pad), img_w - 1)
    y_max = min(int(np.max(pts[:, 1]) + pad), img_h - 1)
    if x_max <= x_min or y_max <= y_min:
        return None
    return (x_min, y_min, x_max, y_max)

def build_bodypart_boxes(kpts_xy, kpts_conf, img_w, img_h, conf_th=0.3):
    """
    kpts_xy: (17,2) pixel coords
    kpts_conf: (17,) confidence
    return: dict(part_name -> xyxy)
    """
    def valid(idxs):
        idxs = [i for i in idxs if kpts_conf[i] >= conf_th]
        if not idxs:
            return np.empty((0, 2), dtype=np.float32)
        return kpts_xy[idxs]

    parts = {}

    # 머리(얼굴): nose/eyes/ears
    head_pts = valid([0, 1, 2, 3, 4])
    box = xywh_from_points(head_pts, img_w, img_h, pad=20)
    if box: parts["head"] = box

    # 몸통: shoulders + hips
    torso_pts = valid([5, 6, 11, 12])
    box = xywh_from_points(torso_pts, img_w, img_h, pad=25)
    if box: parts["torso"] = box

    # 왼팔/오른팔: shoulder-elbow-wrist
    l_arm = valid([5, 7, 9])
    box = xywh_from_points(l_arm, img_w, img_h, pad=20)
    if box: parts["left_arm"] = box

    r_arm = valid([6, 8, 10])
    box = xywh_from_points(r_arm, img_w, img_h, pad=20)
    if box: parts["right_arm"] = box

    # 손(손목 기준): wrist만으로 작은 박스 생성
    for name, idx in [("left_hand", 9), ("right_hand", 10)]:
        if kpts_conf[idx] >= conf_th:
            x, y = kpts_xy[idx]
            x1, y1 = max(int(x - 25), 0), max(int(y - 25), 0)
            x2, y2 = min(int(x + 25), img_w - 1), min(int(y + 25), img_h - 1)
            parts[name] = (x1, y1, x2, y2)

    # 왼다리/오른다리: hip-knee-ankle
    l_leg = valid([11, 13, 15])
    box = xywh_from_points(l_leg, img_w, img_h, pad=25)
    if box: parts["left_leg"] = box

    r_leg = valid([12, 14, 16])
    box = xywh_from_points(r_leg, img_w, img_h, pad=25)
    if box: parts["right_leg"] = box

    # 발(발목 기준)
    for name, idx in [("left_foot", 15), ("right_foot", 16)]:
        if kpts_conf[idx] >= conf_th:
            x, y = kpts_xy[idx]
            x1, y1 = max(int(x - 30), 0), max(int(y - 20), 0)
            x2, y2 = min(int(x + 30), img_w - 1), min(int(y + 20), img_h - 1)
            parts[name] = (x1, y1, x2, y2)

    return parts

def draw_boxes(img, parts, person_id=0):
    for part, (x1, y1, x2, y2) in parts.items():
        cv2.rectangle(img, (x1, y1), (x2, y2), (255, 0, 0), 2)
        cv2.putText(
            img, f"{person_id}:{part}", (x1, max(y1 - 7, 0)),
            cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 0), 2
        )

def run_on_image(image_path, model_path="yolov8n-pose.pt", conf_th=0.3):
    model = YOLO(model_path)

    img = cv2.imread(image_path)
    if img is None:
        raise FileNotFoundError(f"이미지를 읽을 수 없습니다: {image_path}")

    h, w = img.shape[:2]
    results = model(img, conf=0.25, verbose=False)  # 포즈 추론

    r = results[0]
    if r.keypoints is None:
        print("키포인트 결과가 없습니다.")
        return

    # r.keypoints.xy: (num_person, 17, 2) (pixel)
    # r.keypoints.conf: (num_person, 17)
    kxy = r.keypoints.xy.cpu().numpy()
    kcf = r.keypoints.conf.cpu().numpy()

    out = img.copy()
    for i in range(kxy.shape[0]):
        parts = build_bodypart_boxes(kxy[i], kcf[i], w, h, conf_th=conf_th)
        draw_boxes(out, parts, person_id=i)

    cv2.imshow("YOLO Pose Body Parts", out)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

def run_on_webcam(cam_index=0, model_path="yolov8n-pose.pt", conf_th=0.3):
    model = YOLO(model_path)
    cap = cv2.VideoCapture(cam_index)
    if not cap.isOpened():
        raise RuntimeError("웹캠을 열 수 없습니다.")

    while True:
        ret, frame = cap.read()
        if not ret:
            break
        h, w = frame.shape[:2]

        results = model(frame, conf=0.25, verbose=False)
        r = results[0]

        out = frame.copy()
        if r.keypoints is not None:
            kxy = r.keypoints.xy.cpu().numpy()
            kcf = r.keypoints.conf.cpu().numpy()
            for i in range(kxy.shape[0]):
                parts = build_bodypart_boxes(kxy[i], kcf[i], w, h, conf_th=conf_th)
                draw_boxes(out, parts, person_id=i)

        cv2.imshow("YOLO Pose Body Parts (ESC to quit)", out)
        if cv2.waitKey(1) == 27:
            break

    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    # 이미지 테스트
    run_on_image("img/traffic.jpg", conf_th=0.35)

    # 웹캠 실시간 (원하면 주석 해제)
    # run_on_webcam(0, conf_th=0.35)
```

# 3) 이 방식의 한계(정확히 짚고 가기)

* 이 코드는 “부위”를 **키포인트로부터 파생한 박스**입니다.
  즉, **손/발/머리의 정확한 형태(마스크)** 를 얻는 건 아닙니다.
* 손가락, 얼굴 디테일 같은 “세부 부위”는 COCO 17포인트만으로는 부족합니다.

  * 더 정밀하게 하려면: **hand pose 모델 / face landmark / segmentation**을 결합하거나
  * 아예 **부위 클래스(head/hand/foot 등)로 YOLO를 추가 학습**하는 게 정석입니다.

원하시는 “특정부위”가 정확히 뭐예요?

* 예: **머리/얼굴, 손(손바닥), 발, 상반신/하반신, 팔/다리** 정도면 위 방식으로 충분히 됩니다.
* 만약 **“헬멧/장갑/안전조끼 착용 여부”** 같은 산업용 부위/장비까지 원하시면, 그건 **커스텀 데이터로 YOLO 학습** 쪽이 맞습니다.

원하는 부위 목록(예: 얼굴/양손/양발/상체/하체)을 말해주면, 위 코드를 그 부위에 맞춰 **박스 정의/크기/pad/임계값**까지 딱 맞게 튜닝해드릴게요.
