# 🟢 3️⃣ WebSocket + MongoDB + Svelte 통합 실시간 차트 (완성형)

## ✅ backend/main.py

```python
from fastapi import FastAPI, WebSocket
from fastapi.middleware.cors import CORSMiddleware
from pymongo import MongoClient
import random, datetime, asyncio

app = FastAPI()
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"]
)

client = MongoClient("mongodb://localhost:27017")
db = client.chartdb
col = db.data

@app.websocket("/ws")
async def websocket_endpoint(ws: WebSocket):
    await ws.accept()
    while True:
        data = {
            "time": datetime.datetime.now().strftime("%H:%M:%S"),
            "value": random.randint(10,100)
        }
        col.insert_one(data)
        await ws.send_json(data)
        await asyncio.sleep(2)

@app.get("/mongo")
def get_data():
    return list(col.find({}, {"_id":0}))
```

## ✅ frontend/src/App.svelte

```svelte
<script>
import Chart from "chart.js/auto";
import { onMount } from "svelte";

let canvas;
let chart;
let ws;

onMount(() => {
  chart = new Chart(canvas, {
    type: "line",
    data: {
      labels: [],
      datasets: [{ label: "Realtime", data: [] }]
    }
  });

  ws = new WebSocket("ws://localhost:8000/ws");

  ws.onmessage = (event) => {
    const d = JSON.parse(event.data);
    chart.data.labels.push(d.time);
    chart.data.datasets[0].data.push(d.value);
    chart.update();
  };
});
</script>

<canvas bind:this={canvas}></canvas>
```

### A4 실시간 갱신

```svelte
<script>
import Chart from "chart.js/auto";
import { onMount } from "svelte";
let c, chart;
async function getData(){
 return (await fetch("http://localhost:8000/data")).json();
}
onMount(async()=>{
 const d=await getData();
 chart=new Chart(c,{
  type:"line",
  data:{labels:d.map(x=>x.time),
        datasets:[{data:d.map(x=>x.value)}]}
 });
 setInterval(async()=>{
  const nd=await getData();
  chart.data.labels=nd.map(x=>x.time);
  chart.data.datasets[0].data=nd.map(x=>x.value);
  chart.update();
 },2000);
});
</script>
<canvas bind:this={c}></canvas>
```
