# ✅ 버전 1: 전송 속도(FPS, Mbps) 표시 기능

## 📤 Sender (video_sender_v1_speed.py)

```python
import socket, cv2, struct, pickle, threading, time, tkinter as tk
from tkinter import filedialog

HOST="0.0.0.0"; PORT=5000

class Sender:
    def __init__(self, root):
        self.root=root
        tk.Button(root,text="영상 선택",command=self.select).pack()
        tk.Button(root,text="전송 시작",command=self.start).pack()
        self.label=tk.Label(root,text="대기중"); self.label.pack()

    def select(self):
        self.path=filedialog.askopenfilename()

    def start(self):
        threading.Thread(target=self.run,daemon=True).start()

    def run(self):
        s=socket.socket(); s.bind((HOST,PORT)); s.listen(1)
        c,_=s.accept()
        cap=cv2.VideoCapture(self.path)
        start=time.time(); frames=0; bytes_sent=0

        while cap.isOpened():
            ret,frame=cap.read()
            if not ret: break
            data=pickle.dumps(frame)
            c.sendall(struct.pack("Q",len(data))+data)
            frames+=1; bytes_sent+=len(data)

            elapsed=time.time()-start
            fps=frames/elapsed
            mbps=(bytes_sent*8/elapsed)/1e6
            self.label.config(text=f"FPS:{fps:.2f} Mbps:{mbps:.2f}")

        c.close(); s.close()

root=tk.Tk(); Sender(root); root.mainloop()
```

## 📥 Receiver (video_receiver_v1_speed.py)

```python
import socket,struct,pickle,tkinter as tk,time
from PIL import Image,ImageTk
import cv2

PORT=5000

class Receiver:
    def __init__(self,root):
        self.root=root
        tk.Button(root,text="연결",command=self.start).pack()
        self.label=tk.Label(root); self.label.pack()
        self.info=tk.Label(root,text="0 FPS"); self.info.pack()

    def start(self):
        s=socket.socket(); s.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")
        start=time.time(); frames=0

        while True:
            while len(buf)<size: buf+=s.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=s.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]

            frames+=1
            fps=frames/(time.time()-start)
            self.info.config(text=f"FPS:{fps:.2f}")

            img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img

root=tk.Tk(); Receiver(root); root.mainloop()
```

# ✅ 버전 2: 수신 영상 저장 기능

Receiver가 수신 영상을 mp4로 저장

## 📥 Receiver (video_receiver_v2_save.py)

```python
import socket,struct,pickle,tkinter as tk,cv2
from PIL import Image,ImageTk

PORT=5000

class Receiver:
    def __init__(self,root):
        self.root=root
        tk.Button(root,text="연결",command=self.start).pack()
        self.label=tk.Label(root); self.label.pack()

    def start(self):
        s=socket.socket(); s.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")
        out=cv2.VideoWriter("received.mp4",cv2.VideoWriter_fourcc(*'mp4v'),20,(640,480))

        while True:
            while len(buf)<size: buf+=s.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=s.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]

            out.write(frame)
            img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img

root=tk.Tk(); Receiver(root); root.mainloop()
```

# ✅ 버전 3: Play / Pause 버튼 추가

## 📥 Receiver (video_receiver_v3_playpause.py)

```python
import socket,struct,pickle,tkinter as tk
from PIL import Image,ImageTk
import cv2

PORT=5000

class Receiver:
    def __init__(self,root):
        self.root=root; self.running=True
        tk.Button(root,text="연결",command=self.start).pack()
        tk.Button(root,text="Pause",command=self.pause).pack()
        self.label=tk.Label(root); self.label.pack()

    def pause(self):
        self.running=not self.running

    def start(self):
        s=socket.socket(); s.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")

        while True:
            if not self.running: continue
            while len(buf)<size: buf+=s.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=s.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]

            img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img

root=tk.Tk(); Receiver(root); root.mainloop()
```

# ✅ 버전 4: UDP 스트리밍 버전

## 📤 Sender (video_sender_v4_udp.py)

```python
import socket,cv2,pickle

UDP_IP="127.0.0.1"; PORT=5000
sock=socket.socket(socket.AF_INET,socket.SOCK_DGRAM)

cap=cv2.VideoCapture("test.mp4")
while True:
    ret,frame=cap.read()
    if not ret: break
    data=pickle.dumps(frame)
    sock.sendto(data,(UDP_IP,PORT))
```

## 📥 Receiver (video_receiver_v4_udp.py)

```python
import socket,pickle,tkinter as tk
from PIL import Image,ImageTk
import cv2

sock=socket.socket(socket.AF_INET,socket.SOCK_DGRAM)
sock.bind(("",5000))

root=tk.Tk()
label=tk.Label(root); label.pack()

def receive():
    data,_=sock.recvfrom(65536)
    frame=pickle.loads(data)
    img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
    label.config(image=img); label.image=img
    root.after(10,receive)

receive()
root.mainloop()
```

# ✅ 버전 5: 통합 구조 개선 (Sender/Receiver 선택 GUI)

하나의 프로그램에서 역할 선택

```python
import tkinter as tk, socket, cv2, pickle, struct, threading
from PIL import Image,ImageTk

PORT=5000

class App:
    def __init__(self,root):
        tk.Button(root,text="Sender",command=self.sender).pack()
        tk.Button(root,text="Receiver",command=self.receiver).pack()
        self.label=tk.Label(root); self.label.pack()

    def sender(self):
        threading.Thread(target=self.send,daemon=True).start()

    def receiver(self):
        threading.Thread(target=self.recv,daemon=True).start()

    def send(self):
        s=socket.socket(); s.bind(("",PORT)); s.listen(1)
        c,_=s.accept()
        cap=cv2.VideoCapture("test.mp4")
        while True:
            ret,frame=cap.read()
            if not ret: break
            data=pickle.dumps(frame)
            c.sendall(struct.pack("Q",len(data))+data)

    def recv(self):
        c=socket.socket(); c.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")
        while True:
            while len(buf)<size: buf+=c.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=c.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]

            img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img

root=tk.Tk(); App(root); root.mainloop()
```
