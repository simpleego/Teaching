## ✅ 추가 기능

각 프로그램에 **전송(송신) / 수신 진행률 표시(Label + ProgressBar)**

* 송신 서버:
  👉 `전송 프레임 수 / 전체 프레임 수 (%)`
* 수신 서버:
  👉 `수신 프레임 수 / 전체 프레임 수 (%)`

이를 위해:

* 영상의 전체 프레임 수(`CAP_PROP_FRAME_COUNT`) 계산
* 전송/수신 시 카운트 증가
* GUI에 퍼센트 표시

# ✅ 1. 영상 송신 서버 (진행률 표시 포함)

파일명: `video_sender.py`

```python
import socket
import tkinter as tk
from tkinter import filedialog
from tkinter import ttk
import threading
import cv2
import struct
import pickle

HOST = "0.0.0.0"
PORT = 5000

class VideoSenderGUI:
    def __init__(self, root):
        self.root = root
        self.root.title("영상 송신 서버")

        self.video_path = None
        self.total_frames = 0
        self.sent_frames = 0

        tk.Button(root, text="영상 파일 선택", command=self.select_video).pack(pady=5)
        tk.Button(root, text="전송 시작", command=self.start_server).pack(pady=5)

        self.status = tk.Label(root, text="대기 중...")
        self.status.pack()

        self.progress = ttk.Progressbar(root, length=300)
        self.progress.pack(pady=5)

        self.percent_label = tk.Label(root, text="0%")
        self.percent_label.pack()

    def select_video(self):
        self.video_path = filedialog.askopenfilename(
            filetypes=[("Video Files", "*.mp4 *.avi *.mov")]
        )
        self.status.config(text=f"선택된 영상: {self.video_path}")

    def start_server(self):
        if not self.video_path:
            return
        threading.Thread(target=self.server_thread, daemon=True).start()
        self.status.config(text="서버 시작... 연결 대기")

    def server_thread(self):
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server.bind((HOST, PORT))
        server.listen(1)

        conn, addr = server.accept()
        self.status.config(text=f"클라이언트 연결됨: {addr}")

        cap = cv2.VideoCapture(self.video_path)
        self.total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        self.sent_frames = 0

        # 전체 프레임 수 먼저 전송
        conn.sendall(struct.pack("I", self.total_frames))

        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break

            data = pickle.dumps(frame)
            size = struct.pack("Q", len(data))
            conn.sendall(size + data)

            self.sent_frames += 1
            percent = int((self.sent_frames / self.total_frames) * 100)

            self.progress["value"] = percent
            self.percent_label.config(text=f"{percent}% ({self.sent_frames}/{self.total_frames})")

            self.root.update_idletasks()
            cv2.waitKey(30)

        cap.release()
        conn.close()
        server.close()
        self.status.config(text="전송 완료")

if __name__ == "__main__":
    root = tk.Tk()
    app = VideoSenderGUI(root)
    root.mainloop()
```

# ✅ 2. 영상 수신 서버 (진행률 표시 포함)

파일명: `video_receiver.py`

```python
import socket
import tkinter as tk
from tkinter import ttk
from PIL import Image, ImageTk
import threading
import cv2
import struct
import pickle

PORT = 5000

class VideoReceiverGUI:
    def __init__(self, root):
        self.root = root
        self.root.title("영상 수신 서버")

        tk.Label(root, text="서버 IP").pack()
        self.ip_entry = tk.Entry(root)
        self.ip_entry.pack()
        self.ip_entry.insert(0, "127.0.0.1")

        tk.Button(root, text="연결 시작", command=self.start_client).pack(pady=5)

        self.video_label = tk.Label(root)
        self.video_label.pack()

        self.progress = ttk.Progressbar(root, length=300)
        self.progress.pack(pady=5)

        self.percent_label = tk.Label(root, text="0%")
        self.percent_label.pack()

        self.total_frames = 0
        self.received_frames = 0

    def start_client(self):
        threading.Thread(target=self.client_thread, daemon=True).start()

    def client_thread(self):
        client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        client.connect((self.ip_entry.get(), PORT))

        # 전체 프레임 수 수신
        self.total_frames = struct.unpack("I", client.recv(4))[0]
        self.received_frames = 0

        data_buffer = b""
        payload_size = struct.calcsize("Q")

        while True:
            while len(data_buffer) < payload_size:
                packet = client.recv(4096)
                if not packet:
                    return
                data_buffer += packet

            packed_size = data_buffer[:payload_size]
            data_buffer = data_buffer[payload_size:]
            frame_size = struct.unpack("Q", packed_size)[0]

            while len(data_buffer) < frame_size:
                data_buffer += client.recv(4096)

            frame_data = data_buffer[:frame_size]
            data_buffer = data_buffer[frame_size:]

            frame = pickle.loads(frame_data)
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

            img = Image.fromarray(frame)
            img = img.resize((500, 400))
            imgtk = ImageTk.PhotoImage(image=img)

            self.video_label.imgtk = imgtk
            self.video_label.configure(image=imgtk)

            self.received_frames += 1
            percent = int((self.received_frames / self.total_frames) * 100)

            self.progress["value"] = percent
            self.percent_label.config(text=f"{percent}% ({self.received_frames}/{self.total_frames})")

            self.root.update_idletasks()

        client.close()

if __name__ == "__main__":
    root = tk.Tk()
    app = VideoReceiverGUI(root)
    root.mainloop()
```

## ✅ 실행 흐름
