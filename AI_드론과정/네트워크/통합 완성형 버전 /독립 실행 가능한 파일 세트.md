# ✅ 1️⃣ Sender / Receiver 완전 분리 구조화 버전

> 기능: TCP 기반 영상 스트리밍, GUI 분리

## 📤 sender_separated.py

```python
import socket, cv2, pickle, struct, threading, tkinter as tk
from tkinter import filedialog

PORT = 5000

class VideoSender:
    def __init__(self, root):
        root.title("Video Sender")
        self.video_path = None

        tk.Button(root, text="영상 선택", command=self.select_video).pack()
        tk.Button(root, text="송신 시작", command=self.start).pack()
        self.status = tk.Label(root, text="대기중")
        self.status.pack()

    def select_video(self):
        self.video_path = filedialog.askopenfilename(
            filetypes=[("Video Files","*.mp4 *.avi")]
        )

    def start(self):
        threading.Thread(target=self.send_video, daemon=True).start()

    def send_video(self):
        server = socket.socket()
        server.bind(("", PORT))
        server.listen(1)
        conn, addr = server.accept()
        self.status.config(text=f"연결됨: {addr}")

        cap = cv2.VideoCapture(self.video_path)

        while cap.isOpened():
            ret, frame = cap.read()
            if not ret: break

            data = pickle.dumps(frame)
            conn.sendall(struct.pack("Q", len(data)) + data)

        cap.release()
        conn.close()
        server.close()

root = tk.Tk()
VideoSender(root)
root.mainloop()
```

## 📥 receiver_separated.py

```python
import socket, struct, pickle, tkinter as tk
from PIL import Image, ImageTk
import cv2

PORT = 5000

class VideoReceiver:
    def __init__(self, root):
        root.title("Video Receiver")
        tk.Button(root, text="연결", command=self.start).pack()
        self.label = tk.Label(root)
        self.label.pack()

    def start(self):
        sock = socket.socket()
        sock.connect(("127.0.0.1", PORT))

        buffer = b""
        size = struct.calcsize("Q")

        while True:
            while len(buffer) < size:
                buffer += sock.recv(4096)

            frame_size = struct.unpack("Q", buffer[:size])[0]
            buffer = buffer[size:]

            while len(buffer) < frame_size:
                buffer += sock.recv(4096)

            frame_data = buffer[:frame_size]
            buffer = buffer[frame_size:]

            frame = pickle.loads(frame_data)
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

            img = ImageTk.PhotoImage(Image.fromarray(frame))
            self.label.config(image=img)
            self.label.image = img

root = tk.Tk()
VideoReceiver(root)
root.mainloop()
```

```python
"""
Video Sender Program
- 선택한 영상을 TCP로 프레임 단위 전송
"""

import socket, cv2, pickle, struct, threading, tkinter as tk
from tkinter import filedialog

PORT = 5000

class Sender:
    def __init__(self, root):
        root.title("Documented Sender")
        self.video = None

        tk.Button(root, text="영상 선택", command=self.select).pack()
        tk.Button(root, text="전송", command=self.start).pack()

    def select(self):
        self.video = filedialog.askopenfilename()

    def start(self):
        threading.Thread(target=self.run, daemon=True).start()

    def run(self):
        server = socket.socket()
        server.bind(("", PORT))
        server.listen(1)
        conn, addr = server.accept()

        cap = cv2.VideoCapture(self.video)

        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break

            # 프레임 직렬화
            data = pickle.dumps(frame)
            size = struct.pack("Q", len(data))

            # 길이 + 데이터 전송
            conn.sendall(size + data)

        cap.release()
        conn.close()

root = tk.Tk()
Sender(root)
root.mainloop()
```

# ✅ 4️⃣ 안정 스트리밍 고급 버전 (Thread + 오류 처리 + FPS 표시)

## 📤 sender_advanced.py

```python
import socket, cv2, pickle, struct, threading, time, tkinter as tk
from tkinter import filedialog

PORT = 5000

class Sender:
    def __init__(self, root):
        root.title("Advanced Sender")
        self.video=None
        tk.Button(root,text="영상선택",command=self.select).pack()
        tk.Button(root,text="시작",command=self.start).pack()
        self.info=tk.Label(root,text="FPS:0"); self.info.pack()

    def select(self):
        self.video=filedialog.askopenfilename()

    def start(self):
        threading.Thread(target=self.run,daemon=True).start()

    def run(self):
        s=socket.socket()
        s.bind(("",PORT))
        s.listen(1)
        c,_=s.accept()

        cap=cv2.VideoCapture(self.video)
        frames=0
        start=time.time()

        while cap.isOpened():
            ret,frame=cap.read()
            if not ret: break
            data=pickle.dumps(frame)
            c.sendall(struct.pack("Q",len(data))+data)

            frames+=1
            fps=frames/(time.time()-start)
            self.info.config(text=f"FPS:{fps:.2f}")

        cap.release()
        c.close()

root=tk.Tk()
Sender(root)
root.mainloop()
```

## 📥 receiver_advanced.py

```python
import socket, struct, pickle, tkinter as tk, time
from PIL import Image, ImageTk
import cv2

PORT=5000

class Receiver:
    def __init__(self,root):
        root.title("Advanced Receiver")
        tk.Button(root,text="연결",command=self.start).pack()
        self.label=tk.Label(root); self.label.pack()
        self.info=tk.Label(root,text="FPS:0"); self.info.pack()

    def start(self):
        s=socket.socket()
        s.connect(("127.0.0.1",PORT))

        buf=b""; size=struct.calcsize("Q")
        frames=0; start=time.time()

        while True:
            try:
                while len(buf)<size: buf+=s.recv(4096)
                frame_size=struct.unpack("Q",buf[:size])[0]
                buf=buf[size:]

                while len(buf)<frame_size: buf+=s.recv(4096)
                frame=pickle.loads(buf[:frame_size])
                buf=buf[frame_size:]

                frames+=1
                fps=frames/(time.time()-start)

                frame=cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)
                img=ImageTk.PhotoImage(Image.fromarray(frame))
                self.label.config(image=img); self.label.image=img
                self.info.config(text=f"FPS:{fps:.2f}")
            except:
                break

root=tk.Tk()
Receiver(root)
root.mainloop()
```
