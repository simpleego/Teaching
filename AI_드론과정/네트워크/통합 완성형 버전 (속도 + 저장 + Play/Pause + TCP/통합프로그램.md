# ✅ A. 통합 완성형 버전 (속도 + 저장 + Play/Pause + TCP/UDP 선택 + GUI)

파일명: **`video_stream_all_in_one.py`**

기능:

* Sender / Receiver 선택
* TCP / UDP 선택
* FPS / Mbps 표시
* Play / Pause
* 수신 영상 저장

```python
import socket, cv2, pickle, struct, threading, time, tkinter as tk
from tkinter import filedialog
from PIL import Image, ImageTk

PORT = 5000
BUFFER = 4096

class App:
    def __init__(self, root):
        self.root = root
        root.title("All-in-One Video Stream")

        self.mode = tk.StringVar(value="sender")
        self.proto = tk.StringVar(value="tcp")
        self.running = True

        tk.Radiobutton(root, text="Sender", variable=self.mode, value="sender").pack()
        tk.Radiobutton(root, text="Receiver", variable=self.mode, value="receiver").pack()

        tk.Radiobutton(root, text="TCP", variable=self.proto, value="tcp").pack()
        tk.Radiobutton(root, text="UDP", variable=self.proto, value="udp").pack()

        tk.Button(root, text="Video Select", command=self.select_video).pack()
        tk.Button(root, text="Start", command=self.start).pack()
        tk.Button(root, text="Play/Pause", command=self.toggle).pack()

        self.label = tk.Label(root)
        self.label.pack()

        self.info = tk.Label(root, text="FPS:0 Mbps:0")
        self.info.pack()

    def select_video(self):
        self.video = filedialog.askopenfilename()

    def toggle(self):
        self.running = not self.running

    def start(self):
        if self.mode.get() == "sender":
            threading.Thread(target=self.sender, daemon=True).start()
        else:
            threading.Thread(target=self.receiver, daemon=True).start()

    def sender(self):
        sock = socket.socket(socket.AF_INET,
                             socket.SOCK_STREAM if self.proto.get()=="tcp" else socket.SOCK_DGRAM)
        if self.proto.get()=="tcp":
            sock.bind(("", PORT))
            sock.listen(1)
            conn,_ = sock.accept()
        else:
            conn = sock

        cap = cv2.VideoCapture(self.video)
        start = time.time()
        frames=0; bytes_sent=0

        while cap.isOpened():
            if not self.running: continue
            ret, frame = cap.read()
            if not ret: break

            data = pickle.dumps(frame)
            size = struct.pack("Q", len(data))

            if self.proto.get()=="tcp":
                conn.sendall(size+data)
            else:
                conn.sendto(data, ("127.0.0.1", PORT))

            frames+=1; bytes_sent+=len(data)
            elapsed = time.time()-start
            fps = frames/elapsed
            mbps = (bytes_sent*8/elapsed)/1e6
            self.info.config(text=f"FPS:{fps:.2f} Mbps:{mbps:.2f}")

        cap.release()

    def receiver(self):
        sock = socket.socket(socket.AF_INET,
                             socket.SOCK_STREAM if self.proto.get()=="tcp" else socket.SOCK_DGRAM)

        if self.proto.get()=="tcp":
            sock.connect(("127.0.0.1", PORT))
            buf=b""; size=struct.calcsize("Q")
        else:
            sock.bind(("", PORT))

        out = cv2.VideoWriter("received.mp4",
                              cv2.VideoWriter_fourcc(*'mp4v'),20,(640,480))

        start=time.time(); frames=0

        while True:
            if not self.running: continue

            if self.proto.get()=="tcp":
                while len(buf)<size: buf+=sock.recv(BUFFER)
                frame_size=struct.unpack("Q",buf[:size])[0]
                buf=buf[size:]
                while len(buf)<frame_size: buf+=sock.recv(BUFFER)
                frame=pickle.loads(buf[:frame_size])
                buf=buf[frame_size:]
            else:
                data,_=sock.recvfrom(65536)
                frame=pickle.loads(data)

            out.write(frame)
            frames+=1
            fps = frames/(time.time()-start)

            img = ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img
            self.info.config(text=f"FPS:{fps:.2f}")

root=tk.Tk()
App(root)
root.mainloop()
```

# ✅ B. exe 패키징 전용 버전 (PyInstaller)

파일명: **`video_stream_exe.py`**

기능:

* 단순 GUI
* exe 변환용 안정 코드

```python
import socket, cv2, pickle, struct, tkinter as tk
from PIL import Image, ImageTk
import threading

PORT=5000

class App:
    def __init__(self,root):
        root.title("Video Stream EXE")
        tk.Button(root,text="Sender",command=self.sender).pack()
        tk.Button(root,text="Receiver",command=self.receiver).pack()
        self.label=tk.Label(root); self.label.pack()

    def sender(self):
        threading.Thread(target=self.send,daemon=True).start()

    def receiver(self):
        threading.Thread(target=self.recv,daemon=True).start()

    def send(self):
        s=socket.socket()
        s.bind(("",PORT)); s.listen(1)
        c,_=s.accept()
        cap=cv2.VideoCapture(0)

        while True:
            ret,frame=cap.read()
            if not ret: break
            data=pickle.dumps(frame)
            c.sendall(struct.pack("Q",len(data))+data)

    def recv(self):
        c=socket.socket()
        c.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")

        while True:
            while len(buf)<size: buf+=c.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=c.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]
```

# ✅ C. 멀티 클라이언트 서버 버전

파일명: **`video_stream_multiclient.py`**

기능:

* 서버 1대 → 여러 Receiver 동시 스트리밍
* Thread 기반 브로드캐스트

```python
import socket, cv2, pickle, struct, threading, tkinter as tk
from PIL import Image, ImageTk

PORT=5000
clients=[]

class Server:
    def __init__(self,root):
        tk.Button(root,text="Start Server",command=self.start).pack()

    def start(self):
        threading.Thread(target=self.run,daemon=True).start()

    def run(self):
        s=socket.socket()
        s.bind(("",PORT)); s.listen(5)
        threading.Thread(target=self.accept_clients,args=(s,),daemon=True).start()

        cap=cv2.VideoCapture(0)
        while True:
            ret,frame=cap.read()
            if not ret: break
            data=pickle.dumps(frame)
            packet=struct.pack("Q",len(data))+data

            for c in clients:
                try:
                    c.sendall(packet)
                except:
                    clients.remove(c)

    def accept_clients(self,s):
        while True:
            c,_=s.accept()
            clients.append(c)

class Client:
    def __init__(self,root):
        tk.Button(root,text="Connect",command=self.start).pack()
        self.label=tk.Label(root); self.label.pack()

    def start(self):
        threading.Thread(target=self.run,daemon=True).start()

    def run(self):
        c=socket.socket()
        c.connect(("127.0.0.1",PORT))
        buf=b""; size=struct.calcsize("Q")

        while True:
            while len(buf)<size: buf+=c.recv(4096)
            frame_size=struct.unpack("Q",buf[:size])[0]
            buf=buf[size:]
            while len(buf)<frame_size: buf+=c.recv(4096)
            frame=pickle.loads(buf[:frame_size])
            buf=buf[frame_size:]

            img=ImageTk.PhotoImage(Image.fromarray(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)))
            self.label.config(image=img); self.label.image=img

root=tk.Tk()
tk.Button(root,text="Run Server",command=lambda:Server(root)).pack()
tk.Button(root,text="Run Client",command=lambda:Client(root)).pack()
root.mainloop()
```
