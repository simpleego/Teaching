# 1️⃣ 영상 송신 서버 (sender/sender.py)

```python
import cv2
import websocket
import pickle
import struct

ws = websocket.WebSocket()
ws.connect("ws://localhost:8000/ws/sender")

cap = cv2.VideoCapture(0)  # 웹캠 or video.mp4

while True:
    ret, frame = cap.read()
    if not ret:
        break

    data = pickle.dumps(frame)
    size = struct.pack("Q", len(data))

    ws.send_binary(size + data)

cap.release()
ws.close()
```

# 2️⃣ FastAPI 수신 + YOLO 처리 서버 (backend/main.py)

```python
from fastapi import FastAPI, WebSocket
import pickle, struct, cv2
import numpy as np
from ultralytics import YOLO

app = FastAPI()

model = YOLO("yolov8n.pt")

view_clients = []

@app.websocket("/ws/sender")
async def sender_ws(websocket: WebSocket):
    await websocket.accept()
    data_buffer = b""
    payload_size = struct.calcsize("Q")

    try:
        while True:
            data = await websocket.receive_bytes()
            data_buffer += data

            while len(data_buffer) >= payload_size:
                packed_size = data_buffer[:payload_size]
                data_buffer = data_buffer[payload_size:]
                frame_size = struct.unpack("Q", packed_size)[0]

                while len(data_buffer) < frame_size:
                    data_buffer += await websocket.receive_bytes()

                frame_data = data_buffer[:frame_size]
                data_buffer = data_buffer[frame_size:]

                frame = pickle.loads(frame_data)

                # YOLO 객체 탐지
                results = model(frame, stream=False)[0]

                for box in results.boxes:
                    x1,y1,x2,y2 = map(int, box.xyxy[0])
                    label = model.names[int(box.cls[0])]
                    conf = box.conf[0]

                    cv2.rectangle(frame,(x1,y1),(x2,y2),(0,255,0),2)
                    cv2.putText(frame,f"{label} {conf:.2f}",(x1,y1-10),
                                cv2.FONT_HERSHEY_SIMPLEX,0.6,(0,255,0),2)

                # JPEG 인코딩
                _, buffer = cv2.imencode(".jpg", frame)
                jpg_bytes = buffer.tobytes()

                # viewer들에게 전송
                for client in view_clients:
                    await client.send_bytes(jpg_bytes)

    except:
        print("Sender disconnected")

@app.websocket("/ws/viewer")
async def viewer_ws(websocket: WebSocket):
    await websocket.accept()
    view_clients.append(websocket)

    try:
        while True:
            await websocket.receive_text()
    except:
        view_clients.remove(websocket)
```

# 3️⃣ Svelte 프론트엔드 (frontend/src/App.svelte)

```svelte
<script>
  let socket;
  let imgSrc = "";

  socket = new WebSocket("ws://localhost:8000/ws/viewer");
  socket.binaryType = "arraybuffer";

  socket.onmessage = (event) => {
    const blob = new Blob([event.data], { type: "image/jpeg" });
    imgSrc = URL.createObjectURL(blob);
  };
</script>

<h1>AI 영상 스트리밍 모니터</h1>

<img {imgSrc} width="640" height="480" />

<style>
  h1 {
    text-align: center;
  }
  img {
    border: 2px solid black;
  }
</style>
```

실행:

```bash
npm create vite@latest frontend
cd frontend
npm install
npm run dev
```
