## ✅ main.py 전체 코드 (통합 서버)

```python
from fastapi import FastAPI, UploadFile, File
from pydantic import BaseModel
from typing import List

app = FastAPI()

# -------------------
# 데이터 모델
# -------------------
class User(BaseModel):
    id: int = None
    name: str
    age: int

class ChartData(BaseModel):
    labels: List[str]
    values: List[int]

users_db = []
user_id = 1

# -------------------
# 기본 테스트
# -------------------
@app.get("/")
def root():
    return {"message": "FastAPI Server Running"}

# -------------------
# GET 전체 조회
# -------------------
@app.get("/users")
def get_users():
    return users_db

# -------------------
# GET Path Parameter
# -------------------
@app.get("/users/{user_id}")
def get_user(user_id: int):
    for user in users_db:
        if user["id"] == user_id:
            return user
    return {"error": "User not found"}

# -------------------
# POST 생성
# -------------------
@app.post("/users")
def create_user(user: User):
    global user_id
    new_user = user.dict()
    new_user["id"] = user_id
    users_db.append(new_user)
    user_id += 1
    return new_user

# -------------------
# PUT 수정
# -------------------
@app.put("/users/{user_id}")
def update_user(user_id: int, user: User):
    for u in users_db:
        if u["id"] == user_id:
            u["name"] = user.name
            u["age"] = user.age
            return u
    return {"error": "User not found"}

# -------------------
# DELETE 삭제
# -------------------
@app.delete("/users/{user_id}")
def delete_user(user_id: int):
    for u in users_db:
        if u["id"] == user_id:
            users_db.remove(u)
            return {"message": "deleted"}
    return {"error": "User not found"}

# -------------------
# Query Parameter
# -------------------
@app.get("/search")
def search_user(age: int):
    result = [u for u in users_db if u["age"] == age]
    return result

# -------------------
# Chart Data GET
# -------------------
@app.get("/chart-data")
def get_chart_data():
    return {
        "labels": ["Mon","Tue","Wed","Thu","Fri"],
        "values": [10,20,15,30,25]
    }

# -------------------
# Chart Data POST
# -------------------
@app.post("/chart-data")
def post_chart_data(chart: ChartData):
    return {
        "message": "Chart data received",
        "data": chart
    }

# -------------------
# File Upload
# -------------------
@app.post("/upload")
def upload_file(file: UploadFile = File(...)):
    return {
        "filename": file.filename,
        "content_type": file.content_type
    }
```

# ⚠️ 8단계: 자주 발생하는 오류

| 오류   | 원인            | 해결             |
| ---- | ------------- | -------------- |
| 404  | URL 틀림        | 경로 확인          |
| 422  | JSON 구조 틀림    | key 이름 맞추기     |
| 415  | Media Type 오류 | raw → JSON     |
| CORS | 브라우저만 발생      | Postman은 영향 없음 |
