import network
import time
import urequests
import json
import ntptime

ssid = "iptime"
password = "12345678"

# ✅ 본인 위치 위도/경도로 변경
LAT = 36.35   # 예: 대전 근처 (샘플)
LON = 127.38  # 예: 대전 근처 (샘플)

def wifi_connect():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)
    if not wlan.isconnected():
        print("Connecting to WiFi...")
        wlan.connect(ssid, password)
        t0 = time.ticks_ms()
        while not wlan.isconnected():
            if time.ticks_diff(time.ticks_ms(), t0) > 15000:
                raise OSError("WiFi connect timeout")
            time.sleep_ms(300)
    print("Connected:", wlan.ifconfig())

def sync_time():
    # 시간은 출력용 (KST 표시)
    try:
        ntptime.settime()
    except Exception as e:
        print("NTP sync failed:", e)

def now_kst_str():
    t = time.localtime(time.time() + 9*3600)
    return "{:04d}-{:02d}-{:02d} {:02d}:{:02d}:{:02d} KST".format(t[0],t[1],t[2],t[3],t[4],t[5])

def get_current_weather_openmeteo(lat, lon):
    # 현재 온도/습도 + 기상코드
    url = (
        "https://api.open-meteo.com/v1/forecast"
        "?latitude={}&longitude={}"
        "&current=temperature_2m,relative_humidity_2m,weather_code"
        "&timezone=Asia%2FSeoul"
    ).format(lat, lon)

    r = urequests.get(url)
    try:
        if r.status_code != 200:
            raise OSError("HTTP {}".format(r.status_code))
        data = r.json() if hasattr(r, "json") else json.loads(r.text)

        cur = data.get("current", {})
        temp = cur.get("temperature_2m")
        reh  = cur.get("relative_humidity_2m")
        code = cur.get("weather_code")
        tstr = cur.get("time")  # Asia/Seoul 기준 시간 문자열

        return temp, reh, code, tstr
    finally:
        r.close()

def main():
    wifi_connect()
    sync_time()
    print("Now:", now_kst_str())

    try:
        temp, reh, code, tstr = get_current_weather_openmeteo(LAT, LON)
        print("Current (Open-Meteo) time:", tstr)
        print("온도:", temp, "°C")
        print("습도:", reh, "%")
        print("weather_code:", code)
    except Exception as e:
        print("Weather fetch failed:", e)

main()
