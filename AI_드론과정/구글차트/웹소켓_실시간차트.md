아래는 **FastAPI + WebSocket + Svelte + Google Charts**를 이용한
✅ **실시간(Line Chart) 차트 예제**입니다.

서버가 1초마다 데이터를 생성하여 WebSocket으로 전송하고,
Svelte 프론트엔드가 이를 받아 Google Chart를 실시간 갱신합니다.

# ✅ 1. FastAPI WebSocket 서버 (backend/main.py)

```python
from fastapi import FastAPI, WebSocket
from fastapi.middleware.cors import CORSMiddleware
import asyncio
import random
import datetime

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.websocket("/ws/chart")
async def websocket_endpoint(websocket: WebSocket):
    await websocket.accept()

    while True:
        now = datetime.datetime.now().strftime("%H:%M:%S")
        value = random.randint(0, 100)

        data = {
            "time": now,
            "value": value
        }

        await websocket.send_json(data)
        await asyncio.sleep(1)
```

# ✅ 2. Svelte + Google Chart (frontend/App.svelte)

```svelte
<script>
  import { onMount } from "svelte";

  let chart;
  let dataTable;
  let ws;

  onMount(() => {
    google.charts.load("current", { packages: ["corechart"] });
    google.charts.setOnLoadCallback(initChart);
  });

  function initChart() {
    dataTable = new google.visualization.DataTable();
    dataTable.addColumn("string", "Time");
    dataTable.addColumn("number", "Value");

    chart = new google.visualization.LineChart(
      document.getElementById("chart")
    );

    connectWebSocket();
  }

  function connectWebSocket() {
    ws = new WebSocket("ws://localhost:8000/ws/chart");

    ws.onmessage = (event) => {
      const msg = JSON.parse(event.data);

      dataTable.addRow([msg.time, msg.value]);

      // 최대 10개 데이터 유지
      if (dataTable.getNumberOfRows() > 10) {
        dataTable.removeRow(0);
      }

      drawChart();
    };

    ws.onclose = () => {
      console.log("WebSocket closed");
    };
  }

  function drawChart() {
    const options = {
      title: "Realtime WebSocket Chart",
      curveType: "function",
      legend: { position: "bottom" }
    };

    chart.draw(dataTable, options);
  }
</script>

<svelte:head>
  <script src="https://www.gstatic.com/charts/loader.js"></script>
</svelte:head>

<h2>WebSocket Realtime Chart</h2>
<div id="chart" style="width:700px; height:400px;"></div>
```
