# ✅ 해결 방법 1: 자동 재접속(Reconnect) 로직 추가 (가장 중요)

```python
import cv2
import time

url = "rtsp://192.168.0.10:8554/stream"

while True:
    print("스트림 연결 시도...")
    cap = cv2.VideoCapture(url)

    if not cap.isOpened():
        print("연결 실패, 5초 후 재시도")
        time.sleep(5)
        continue

    while True:
        ret, frame = cap.read()

        if not ret:
            print("프레임 수신 실패, 재연결 시도")
            cap.release()
            break

        cv2.imshow("Stream", frame)
        if cv2.waitKey(1) == 27:
            cap.release()
            cv2.destroyAllWindows()
            exit()

    time.sleep(2)
```

# ✅ 해결 방법 3: RTSP 옵션 튜닝 (지연 감소 + 안정성)

```python
url = "rtsp://192.168.0.10:8554/stream?tcp"

cap = cv2.VideoCapture(url, cv2.CAP_FFMPEG)
```

또는 GStreamer 사용:

```python
gst = (
    "rtspsrc location=rtsp://192.168.0.10:8554/stream latency=0 ! "
    "decodebin ! videoconvert ! appsink"
)

cap = cv2.VideoCapture(gst, cv2.CAP_GSTREAMER)
```

# 🎯 실무 권장 구조 (안정 버전)

```python
import cv2, time

url = "rtsp://192.168.0.10:8554/stream"

while True:
    cap = cv2.VideoCapture(url)
    cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)

    if not cap.isOpened():
        print("연결 실패, 재시도...")
        time.sleep(3)
        continue

    while True:
        ret, frame = cap.read()
        if not ret:
            print("스트림 끊김")
            break

        cv2.imshow("Drone Stream", frame)
        if cv2.waitKey(1) == 27:
            cap.release()
            cv2.destroyAllWindows()
            exit()

    cap.release()
    time.sleep(2)
```
