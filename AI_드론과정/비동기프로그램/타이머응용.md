# ✅ 1. 기본 스톱워치 앱 (Start / Stop / Reset)

```html
<h2>1. Stopwatch</h2>
<div id="stopwatch">0</div>
<button onclick="startStopwatch()">Start</button>
<button onclick="stopStopwatch()">Stop</button>
<button onclick="resetStopwatch()">Reset</button>

<script>
let sw = 0;
let swTimer = null;

function startStopwatch() {
  if (swTimer) return;
  swTimer = setInterval(() => {
    sw++;
    document.getElementById("stopwatch").innerText = sw;
  }, 1000);
}

function stopStopwatch() {
  clearInterval(swTimer);
  swTimer = null;
}

function resetStopwatch() {
  stopStopwatch();
  sw = 0;
  document.getElementById("stopwatch").innerText = sw;
}
</script>
```

# ✅ 2. 카운트다운 타이머 앱

```html
<h2>2. Countdown Timer</h2>
<input type="number" id="countInput" placeholder="초 입력">
<button onclick="startCountdown()">Start</button>
<div id="countdown"></div>

<script>
let cdTimer;

function startCountdown() {
  let time = Number(document.getElementById("countInput").value);

  cdTimer = setInterval(() => {
    document.getElementById("countdown").innerText = time;

    if (time <= 0) {
      clearInterval(cdTimer);
      alert("타이머 종료!");
    }
    time--;
  }, 1000);
}
</script>
```

# ✅ 3. Pomodoro 타이머 (25분 집중 / 5분 휴식)

```html
<h2>3. Pomodoro Timer</h2>
<div id="pomodoro">25:00</div>
<button onclick="startPomodoro()">Start</button>

<script>
let pomoTime = 1500; // 25분
let pomoTimer;

function startPomodoro() {
  pomoTimer = setInterval(() => {
    let min = String(Math.floor(pomoTime / 60)).padStart(2,"0");
    let sec = String(pomoTime % 60).padStart(2,"0");
    document.getElementById("pomodoro").innerText = `${min}:${sec}`;

    if (pomoTime <= 0) {
      clearInterval(pomoTimer);
      alert("휴식 시간!");
    }
    pomoTime--;
  }, 1000);
}
</script>
```

# ✅ 4. 다중 타이머 관리 앱 (여러 개 동시 실행)

```html
<h2>4. Multi Timer</h2>
<button onclick="addTimer()">타이머 추가</button>
<div id="timers"></div>

<script>
let timerId = 0;

function addTimer() {
  const id = timerId++;
  let count = 0;

  const div = document.createElement("div");
  div.id = "timer-" + id;
  div.innerHTML = `
    Timer ${id}: <span>0</span>
    <button onclick="stopMultiTimer(${id})">Stop</button>
  `;
  document.getElementById("timers").appendChild(div);

  const interval = setInterval(() => {
    count++;
    div.querySelector("span").innerText = count;
  }, 1000);

  div.dataset.interval = interval;
}

function stopMultiTimer(id) {
  const div = document.getElementById("timer-" + id);
  clearInterval(div.dataset.interval);
}
</script>
```

# ✅ 5. 예약 실행 스케줄러 앱 (setTimeout 활용)

```html
<h2>5. Scheduler</h2>
<input type="text" id="task" placeholder="작업 이름">
<input type="number" id="delay" placeholder="초 후 실행">
<button onclick="scheduleTask()">예약</button>
<ul id="taskList"></ul>

<script>
function scheduleTask() {
  const task = document.getElementById("task").value;
  const delay = Number(document.getElementById("delay").value) * 1000;

  const li = document.createElement("li");
  li.innerText = `${task} 예약됨`;
  document.getElementById("taskList").appendChild(li);

  setTimeout(() => {
    li.innerText = `${task} 실행 완료`;
  }, delay);
}
</script>
```
